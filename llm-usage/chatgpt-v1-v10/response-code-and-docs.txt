from time import perf_counter

from playerTeamName import PlayerTeamName

SIZE = 19
EMPTY = -1
WALL = 2
STRIDE = SIZE + 2
AXES = (1, STRIDE, STRIDE + 1, STRIDE - 1)
DIRECTIONS = AXES + tuple(-d for d in AXES)
CELLS = tuple((r + 1) * STRIDE + c + 1 for r in range(SIZE) for c in range(SIZE))
MATE = 10_000_000
WINDOW_WEIGHTS = (0, 2, 30, 450, 8_000, 0)


class SmartPlayerTeamName(PlayerTeamName):
    def __init__(self, color):
        super().__init__(color)
        if color not in (0, 1):
            raise ValueError("color must be 0 (black) or 1 (white)")
        self._lines = self._build_lines()

    def take_turn(self, board, time):
        started = perf_counter()
        self._grid = [WALL] * (STRIDE * STRIDE)
        for r in range(SIZE):
            for c in range(SIZE):
                self._grid[(r + 1) * STRIDE + c + 1] = int(board[r][c])
        empty = [p for p in CELLS if self._grid[p] == EMPTY]
        if not empty:
            raise ValueError("No legal move: the board is full.")
        if len(empty) == SIZE * SIZE:
            return SIZE // 2, SIZE // 2

        candidates = self._candidates()
        best = min(candidates, key=self._centre_distance)
        remaining = float(time) / 1000 if time != -1 else None
        if remaining is not None and remaining <= 0.015:
            return self._coordinates(best)
        budget = 1.0 if remaining is None else min(1.0, remaining / 30, max(0, remaining - 0.02))
        self._deadline = started + budget
        self._line_cache = {}
        self.last_depth = 0
        self.last_nodes = 0

        try:
            ranked = self._rank_moves(self.color)
            if ranked:
                best = ranked[0][1]
            if ranked and ranked[0][0] >= MATE:
                return self._coordinates(best)
            for depth in range(1, 5):
                ordered = sorted(ranked[:14], key=lambda item: item[1] != best)
                depth_best, depth_score = best, -2 * MATE
                alpha = -2 * MATE
                for _, move in ordered:
                    self._check_time()
                    captured = self._make_move(move, self.color)
                    try:
                        outcome = self._outcome(move, self.color, captured)
                        if outcome is not None:
                            score = outcome * MATE
                        else:
                            score = -self._search(1 - self.color, depth - 1, -2 * MATE, -alpha, 1)
                    finally:
                        self._undo_move(move, self.color, captured)
                    if score > depth_score:
                        depth_best, depth_score = move, score
                    alpha = max(alpha, score)
                best = depth_best
                self.last_depth = depth
                if abs(depth_score) > MATE - 1000:
                    break
        except TimeoutError:
            pass
        return self._coordinates(best)

    def _search(self, color, depth, alpha, beta, ply):
        self._check_time()
        self.last_nodes += 1
        if not any(self._grid[p] == EMPTY for p in CELLS):
            return 0
        if depth == 0:
            return self._evaluate(color)
        moves = self._rank_moves(color)
        if moves[0][0] >= MATE:
            return MATE - ply
        best = -2 * MATE
        for _, move in moves[:10]:
            self._check_time()
            captured = self._make_move(move, color)
            try:
                outcome = self._outcome(move, color, captured)
                if outcome is not None:
                    score = outcome * (MATE - ply)
                else:
                    score = -self._search(1 - color, depth - 1, -beta, -alpha, ply + 1)
            finally:
                self._undo_move(move, color, captured)
            best = max(best, score)
            alpha = max(alpha, score)
            if alpha >= beta:
                break
        return best

    def _rank_moves(self, color):
        opponent = 1 - color
        result, winning, threats = [], [], []
        for move in self._candidates():
            self._check_time()
            captured = self._make_move(move, color)
            try:
                outcome = self._outcome(move, color, captured)
                attack = self._local_score(move, color) + 600 * len(captured)
                attack -= 900 * self._vulnerable_pairs(move, color)
            finally:
                self._undo_move(move, color, captured)
            if outcome == 1:
                winning.append((MATE, move))
            captured = self._make_move(move, opponent)
            try:
                other_outcome = self._outcome(move, opponent, captured)
                defence = self._local_score(move, opponent) + 600 * len(captured)
            finally:
                self._undo_move(move, opponent, captured)
            if other_outcome == 1:
                threats.append(move)
            score = attack + 0.95 * defence - self._centre_distance(move) * 0.1
            if outcome == -1:
                score = -MATE
            elif outcome == 0:
                score = -50
            result.append((score, move))
        if winning:
            return sorted(winning, reverse=True)
        if threats:
            adjusted = []
            safe_moves = []
            for score, move in result:
                self._check_time()
                captured = self._make_move(move, color)
                try:
                    outcome = self._outcome(move, color, captured)
                    safe = outcome == 0
                    if outcome is None:
                        safe = True
                        for reply in threats:
                            if self._grid[reply] != EMPTY:
                                continue
                            taken = self._make_move(reply, opponent)
                            try:
                                if self._outcome(reply, opponent, taken) == 1:
                                    safe = False
                            finally:
                                self._undo_move(reply, opponent, taken)
                            if not safe:
                                break
                    adjusted.append((score if safe else score - 2 * MATE, move))
                    if safe:
                        safe_moves.append((score, move))
                finally:
                    self._undo_move(move, color, captured)
            result = safe_moves or adjusted
        return sorted(result, reverse=True)

    def _candidates(self):
        grid = self._grid
        near = set()
        for p in CELLS:
            if grid[p] in (0, 1):
                r, c = self._coordinates(p)
                for rr in range(max(0, r - 2), min(SIZE, r + 3)):
                    for cc in range(max(0, c - 2), min(SIZE, c + 3)):
                        q = (rr + 1) * STRIDE + cc + 1
                        if grid[q] == EMPTY:
                            near.add(q)
        return sorted(near) or [p for p in CELLS if grid[p] == EMPTY]

    def _make_move(self, move, color):
        grid = self._grid
        grid[move] = color
        opponent = 1 - color
        captured = []
        for direction in DIRECTIONS:
            a, b, end = move + direction, move + 2 * direction, move + 3 * direction
            if 0 <= end < len(grid) and grid[a] == opponent and grid[b] == opponent and grid[end] == color:
                captured.extend((a, b))
        for p in captured:
            grid[p] = EMPTY
        return captured

    def _undo_move(self, move, color, captured):
        self._grid[move] = EMPTY
        for p in captured:
            self._grid[p] = 1 - color

    def _exact_five_at(self, move, color):
        if self._grid[move] != color:
            return False
        for axis in AXES:
            length = 1
            for direction in (axis, -axis):
                p = move + direction
                while 0 <= p < len(self._grid) and self._grid[p] == color:
                    length += 1
                    p += direction
            if length == 5:
                return True
        return False

    def _outcome(self, move, color, captured):
        mine = self._exact_five_at(move, color)
        opponent = 1 - color
        theirs = False
        for removed in captured:
            for direction in DIRECTIONS:
                p = removed + direction
                if 0 <= p < len(self._grid) and self._grid[p] == opponent:
                    if self._exact_five_at(p, opponent):
                        theirs = True
                        break
            if theirs:
                break
        if mine and theirs:
            return 0
        if mine:
            return 1
        if theirs:
            return -1
        return None

    def _local_score(self, move, color):
        grid = self._grid
        total = 0
        for axis in AXES:
            length, open_ends = 1, 0
            for direction in (axis, -axis):
                p = move + direction
                while 0 <= p < len(grid) and grid[p] == color:
                    length += 1
                    p += direction
                if 0 <= p < len(grid) and grid[p] == EMPTY:
                    open_ends += 1
            if length == 4:
                total += 45_000 if open_ends == 2 else 6_000 if open_ends else 0
            elif length == 3:
                total += 3_000 if open_ends == 2 else 250 if open_ends else 0
            elif length == 2:
                total += 120 if open_ends == 2 else 20 if open_ends else 0
            for offset in range(5):
                start = move - offset * axis
                end = start + 4 * axis
                if not (0 <= start < len(grid) and 0 <= end < len(grid)):
                    continue
                cells = [grid[start + i * axis] for i in range(5)]
                if any(value not in (EMPTY, color) for value in cells):
                    continue
                before, after = start - axis, end + axis
                if 0 <= before < len(grid) and grid[before] == color:
                    continue
                if 0 <= after < len(grid) and grid[after] == color:
                    continue
                total += WINDOW_WEIGHTS[cells.count(color)]
        return total

    def _vulnerable_pairs(self, move, color):
        count, grid = 0, self._grid
        for direction in DIRECTIONS:
            a, before, after = move + direction, move - direction, move + 2 * direction
            if not (0 <= before < len(grid) and 0 <= after < len(grid)):
                continue
            if grid[a] == color and {grid[before], grid[after]} == {1 - color, EMPTY}:
                count += 1
        return count

    def _evaluate(self, color):
        total = 0
        for line in self._lines:
            values = tuple(self._grid[p] for p in line)
            if values not in self._line_cache:
                self._line_cache[values] = self._score_line(values, 0) - self._score_line(values, 1)
            total += self._line_cache[values]
        return total if color == 0 else -total

    def _score_line(self, values, color):
        total = 0
        for start in range(len(values) - 4):
            window = values[start:start + 5]
            if any(value not in (EMPTY, color) for value in window):
                continue
            if start and values[start - 1] == color:
                continue
            if start + 5 < len(values) and values[start + 5] == color:
                continue
            count = window.count(color)
            total += WINDOW_WEIGHTS[count]
        i = 0
        while i < len(values):
            if values[i] != color:
                i += 1
                continue
            start = i
            while i < len(values) and values[i] == color:
                i += 1
            length = i - start
            ends = int(start > 0 and values[start - 1] == EMPTY) + int(i < len(values) and values[i] == EMPTY)
            if length == 4 and ends == 2:
                total += 30_000
            elif length == 3 and ends == 2:
                total += 2_000
            elif length == 2 and ends == 2:
                total += 80
        for i in range(len(values) - 3):
            pair = values[i:i + 4]
            if pair in ((1 - color, color, color, EMPTY), (EMPTY, color, color, 1 - color)):
                total -= 700
        return total

    def _build_lines(self):
        lines = []
        for axis in AXES:
            for p in CELLS:
                r, c = self._coordinates(p - axis)
                if 0 <= r < SIZE and 0 <= c < SIZE:
                    continue
                line, q = [], p
                while True:
                    r, c = self._coordinates(q)
                    if not (0 <= r < SIZE and 0 <= c < SIZE):
                        break
                    line.append(q)
                    q += axis
                if len(line) >= 4:
                    lines.append(tuple(line))
        return tuple(lines)

    def _coordinates(self, move):
        return move // STRIDE - 1, move % STRIDE - 1

    def _centre_distance(self, move):
        r, c = self._coordinates(move)
        return abs(r - SIZE // 2) + abs(c - SIZE // 2)

    def _check_time(self):
        if perf_counter() >= self._deadline:
            raise TimeoutError

def grid():
    return [[-1] * SIZE for _ in range(SIZE)]


def has_five(board, color):
    for r in range(SIZE):
        for c in range(SIZE):
            if board[r][c] != color:
                continue
            for dr, dc in AXES:
                pr, pc = r - dr, c - dc
                if 0 <= pr < SIZE and 0 <= pc < SIZE and board[pr][pc] == color:
                    continue
                n, rr, cc = 0, r, c
                while 0 <= rr < SIZE and 0 <= cc < SIZE and board[rr][cc] == color:
                    n += 1
                    rr, cc = rr + dr, cc + dc
                if n == 5:
                    return True
    return False


def oracle(board, move, color):
    after = [row[:] for row in board]
    r, c = move
    after[r][c] = color
    removed = set()
    for dr, dc in AXES:
        for sign in (-1, 1):
            rr, cc = r + sign * 3 * dr, c + sign * 3 * dc
            if not (0 <= rr < SIZE and 0 <= cc < SIZE):
                continue
            a = (r + sign * dr, c + sign * dc)
            b = (r + sign * 2 * dr, c + sign * 2 * dc)
            if after[a[0]][a[1]] == after[b[0]][b[1]] == 1 - color and after[rr][cc] == color:
                removed.update((a, b))
    for rr, cc in removed:
        after[rr][cc] = -1
    mine, theirs = has_five(after, color), has_five(after, 1 - color)
    result = 0 if mine and theirs else 1 if mine else -1 if theirs else None
    return after, removed, result


def set_position(agent, board):
    agent._grid = [WALL] * (STRIDE * STRIDE)
    for r in range(SIZE):
        for c in range(SIZE):
            agent._grid[(r + 1) * STRIDE + c + 1] = board[r][c]
    agent._deadline = float("inf")


def check_simulation(agent, board, move, color):
    set_position(agent, board)
    original = agent._grid[:]
    p = (move[0] + 1) * STRIDE + move[1] + 1
    taken = agent._make_move(p, color)
    expected, removed, result = oracle(board, move, color)
    actual = [[agent._grid[(r + 1) * STRIDE + c + 1] for c in range(SIZE)] for r in range(SIZE)]
    assert actual == expected, (move, color, "capture mismatch")
    assert {agent._coordinates(q) for q in taken} == removed
    assert agent._outcome(p, color, taken) == result, (move, color, "outcome mismatch")
    agent._undo_move(p, color, taken)
    assert agent._grid == original, "undo mismatch"

def play(opponent_class, agent_color, seed, board_class):
    random.seed(seed)
    board = board_class()
    players = [None, None]
    players[agent_color] = SmartPlayerTeamName(agent_color)
    players[1 - agent_color] = opponent_class(1 - agent_color)
    remaining = [60.0, 60.0]
    turns = [0, 0]
    current = 0
    for ply in range(1200):
        before = board.GetBoard()
        snapshot = [row[:] for row in before]
        started = perf_counter()
        try:
            move = players[current].take_turn(before, remaining[current] * 1000)
        except Exception as exc:
            return {"result": "loss" if current == agent_color else "win", "reason": f"player {current} exception: {type(exc).__name__}: {exc}", "plies": ply}
        remaining[current] -= perf_counter() - started
        turns[current] += 1
        if remaining[current] < 0 or before != snapshot:
            return {"result": "loss" if current == agent_color else "win", "reason": "timeout or input mutation", "plies": ply}
        if not (isinstance(move, tuple) and len(move) == 2 and all(type(x) is int for x in move) and board.ValidPosition(*move)):
            return {"result": "loss" if current == agent_color else "win", "reason": "illegal move", "plies": ply}
        board.PlaceStone(*move, current)
        board.TakeStone(*move, current)
        wins = [board.CheckOmok(color) for color in (0, 1)]
        result = "draw" if all(wins) else "win" if wins[agent_color] else "loss" if wins[1 - agent_color] else "draw" if board.FullBoard() else None
        if result:
            return {"result": result, "reason": "board result", "plies": ply + 1, "agent_seconds": round(60 - remaining[agent_color], 4), "opponent_seconds": round(60 - remaining[1 - agent_color], 4), "agent_turns": turns[agent_color]}
        current = 1 - current
    return {"result": "unfinished", "reason": "testing ply cap (not a game draw)", "plies": 1200}

# 이번 GameAgent가 하는 일

기존 프로젝트의 `GameManager`는 실제 게임 진행을 맡습니다. 새 `SmartPlayerTeamName`은 자기 차례에 받은 판을 보고 **어디에 둘지 결정하는 일**을 맡습니다.

```python
agent = SmartPlayerTeamName(0)
row, col = agent.take_turn(board, 60000)
```

`0`은 흑, `1`은 백입니다. `60000`은 이 플레이어에게 게임 전체에서 남은 60,000ms입니다. 입력받은 `board`는 바꾸지 않고 `(행, 열)`만 반환합니다.

## 수를 고르는 과정

1. 빈 칸 전체를 무작정 탐색하지 않고, 기존 돌에서 두 칸 이내인 위치를 후보로 모읍니다.
2. 후보 위치에 내 돌을 놓았다고 가정하고, 포획할 돌을 제거합니다.
3. 이 상태가 내 승리·상대 승리·동시 승리인지 확인합니다.
4. 같은 위치에 상대가 두었을 때의 위협도 계산합니다.
5. 상대에게 즉시 승리 수가 있으면 그 위협을 없애는 수를 우선 검토합니다. 상대의 두 돌을 잡아 공격 모양을 깨는 방어도 포함합니다.
6. 유망한 수부터 상대의 응수와 이후의 내 수를 탐색합니다.
7. 시간 예산이 끝나면 마지막으로 계산을 마친 결과를 반환합니다.

## SimplePlayer보다 추가된 점

기존 SimplePlayer는 주로 지금 한 수를 둔 직후의 모양에 점수를 줍니다. 이번 버전은 상대도 가장 좋은 응수를 고른다고 가정하고 이후의 판까지 비교합니다. 당장의 모양이 좋아도 상대가 다음 차례에 이길 수 있으면 탐색에서 불리하게 평가됩니다. 다만 깊이와 후보 수를 제한하므로 놓치는 수가 남아 있습니다.

## 반드시 이해해야 할 네 가지

**가상 착수와 원상 복구:** `_make_move()`는 개인 복사본에 돌을 놓고 포획합니다. `_undo_move()`는 그 돌을 빼고 잡았던 돌을 돌려놓습니다. 이것을 반복해야 서로 다른 후보를 같은 출발점에서 비교할 수 있습니다. `try/finally`는 계산 도중 시간 제한으로 빠져나와도 판을 원상 복구하게 합니다.

**평가 함수:** `_local_score()`는 후보 정렬에 쓰고 `_evaluate()`는 탐색 끝의 판 전체를 평가합니다. 다섯 칸 구간에 내 돌이 얼마나 모였는지, 양 끝이 열린 연속 모양인지, 잡힐 수 있는 두 돌이 있는지 등을 숫자로 나타냅니다. 점수 상수는 증명된 정답이 아니라 대결로 조정할 요소입니다.

**상대 응수 탐색:** `_search()`는 현재 차례의 플레이어 관점에서 점수를 반환합니다. 상대에게 좋은 점수는 나에게 나쁜 점수이므로 반환값의 부호를 바꿉니다. 알파베타는 이미 확인한 선택보다 나아질 가능성이 없는 가지를 계산하지 않는 방법입니다.

**반복 심화와 시간:** 한 수 깊이 탐색을 마친 뒤 두 수, 세 수로 깊이를 늘립니다. 세 수 탐색 도중 시간이 끝나도 두 수 탐색의 완성된 결과를 쓸 수 있습니다. 최대 깊이 네 수는 목표이고 매 차례 보장하는 깊이는 아닙니다.

## 왜 포획 이후의 상대 승리도 확인하나

상대의 여섯 돌이 연속되어 있으면 그 자체는 승리가 아닙니다. 그런데 내가 그 끝 돌을 잡아 다섯 돌만 남기면 상대가 이깁니다. 같은 수로 내 다섯 돌까지 완성되면 무승부입니다. 그래서 포획은 항상 공격상 이득이라고 가정하면 안 됩니다.
