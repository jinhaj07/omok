from time import perf_counter
from collections import OrderedDict

from playerJangJinha import PlayerJangJinha

SIZE = 19
EMPTY = -1
WALL = 2
STRIDE = SIZE + 2
AXES = (1, STRIDE, STRIDE + 1, STRIDE - 1)
DIRECTIONS = AXES + tuple(-d for d in AXES)
CELLS = tuple((r + 1) * STRIDE + c + 1 for r in range(SIZE) for c in range(SIZE))
MATE = 10_000_000
WINDOW_WEIGHTS = (0, 2, 30, 450, 8_000, 0)
MAX_DEPTH = 6
QUIESCENCE_DEPTH = 6
ROOT_WIDTH = 20
INNER_WIDTH = 12
FORCING_WIDTH = 8
POSITION_CACHE_LIMIT = 20_000
LINE_CACHE_LIMIT = 40_000


class SmartPlayerJangJinha(PlayerJangJinha):
    def __init__(self, color):
        super().__init__(color)
        if color not in (0, 1):
            raise ValueError("color must be 0 (black) or 1 (white)")
        self._lines = self._build_lines()
        neighbors = [[] for _ in range(STRIDE * STRIDE)]
        for p in CELLS:
            r, c = self._coordinates(p)
            neighbors[p] = tuple((rr + 1) * STRIDE + cc + 1
                                 for rr in range(max(0, r - 2), min(SIZE, r + 3))
                                 for cc in range(max(0, c - 2), min(SIZE, c + 3)))
        self._neighbors = tuple(neighbors)
        self._indexed_grid = None
        self._line_cache = OrderedDict()
        self._q_depth = QUIESCENCE_DEPTH
        self.last_depth = 0
        self.last_nodes = 0
        self.last_qnodes = 0
        self.last_extension = 0
        self._tactical_cache = {}
        self._defence_cache = {}
        self._forcing_cache = {}
        self.last_tt_hits = 0
        self.last_rank_hits = 0
        self.last_move_score_hits = 0
        self.last_winning_hits = 0
        self.last_line_hits = 0
        self.last_make_calls = 0

    def take_turn(self, board, time):
        started = perf_counter()
        self._grid = [WALL] * (STRIDE * STRIDE)
        for r in range(SIZE):
            for c in range(SIZE):
                self._grid[(r + 1) * STRIDE + c + 1] = int(board[r][c])
        empty = [p for p in CELLS if self._grid[p] == EMPTY]
        if not empty:
            raise ValueError("No legal move: the board is full.")
        if len(empty) == SIZE * SIZE:
            return SIZE // 2, SIZE // 2

        candidates = self._candidates()
        best = min(candidates, key=self._centre_distance)
        remaining = float(time) / 1000 if time != -1 else None
        if remaining is not None and remaining <= 0.015:
            return self._coordinates(best)
        budget = 1.0 if remaining is None else min(1.0, remaining / 30, max(0, remaining - 0.02))
        self._deadline = started + budget
        self.last_depth = 0
        self.last_nodes = 0
        self.last_qnodes = 0
        self.last_extension = 0
        self.last_tt_hits = 0
        self.last_rank_hits = 0
        self.last_move_score_hits = 0
        self.last_winning_hits = 0
        self.last_line_hits = 0
        self.last_make_calls = 0

        try:
            ranked = self._rank_moves(self.color)
            if ranked:
                best = ranked[0][1]
            if ranked and ranked[0][0] >= MATE:
                return self._coordinates(best)
            for depth in range(1, MAX_DEPTH + 1):
                # Compare root candidates cheaply before extending tactical lines.
                # At zero extension, immediate wins and safe defences still run.
                self._q_depth = 0 if depth == 1 else QUIESCENCE_DEPTH
                selected = self._select_moves(ranked, self.color, ROOT_WIDTH)
                ordered = sorted(selected, key=lambda item: item[1] != best)
                depth_best, depth_score = best, -2 * MATE
                alpha = -2 * MATE
                for _, move in ordered:
                    self._check_time()
                    captured = self._make_move(move, self.color)
                    try:
                        outcome = self._outcome(move, self.color, captured)
                        if outcome is not None:
                            score = outcome * MATE
                        else:
                            score = -self._search(1 - self.color, depth - 1, -2 * MATE, -alpha, 1)
                    finally:
                        self._undo_move(move, self.color, captured)
                    if score > depth_score:
                        depth_best, depth_score = move, score
                        if depth == 1:
                            best = depth_best
                    alpha = max(alpha, score)
                    if depth == 1 and score > MATE - 1000:
                        self.last_depth = 1
                        return self._coordinates(move)
                best = depth_best
                self.last_depth = depth
        except TimeoutError:
            pass
        finally:
            self._q_depth = QUIESCENCE_DEPTH
        return self._coordinates(best)

    def _search(self, color, depth, alpha, beta, ply):
        self._check_time()
        self._ensure_position()
        self.last_nodes += 1
        # First-pass values must never replace deeper tactical-search values.
        key = (self._position_key(color), depth, ply, self._q_depth)
        cached = self._search_cache.get(key)
        if cached is not None:
            value, flag = cached
            if flag == 0 or flag == 1 and value >= beta or flag == -1 and value <= alpha:
                self.last_tt_hits += 1
                return value
        value = self._search_uncached(color, depth, alpha, beta, ply)
        self._remember(self._search_cache, key, (value, self._bound(value, alpha, beta)))
        return value

    def _search_uncached(self, color, depth, alpha, beta, ply):
        if self._empty_count == 0:
            return 0
        if depth == 0:
            return self._quiescence(color, alpha, beta, ply, self._q_depth)
        wins, draws, threats = self._tactical_state(color)
        if wins:
            return MATE - ply
        if threats and not self._defending_moves(color):
            return -MATE + ply + 1
        moves = self._rank_moves(color)
        if moves[0][0] >= MATE:
            return MATE - ply
        best = -2 * MATE
        for _, move in self._select_moves(moves, color, INNER_WIDTH):
            self._check_time()
            captured = self._make_move(move, color)
            try:
                outcome = self._outcome(move, color, captured)
                if outcome is not None:
                    score = outcome * (MATE - ply)
                else:
                    score = -self._search(1 - color, depth - 1, -beta, -alpha, ply + 1)
            finally:
                self._undo_move(move, color, captured)
            best = max(best, score)
            alpha = max(alpha, score)
            if alpha >= beta:
                break
        return best

    def _rank_moves(self, color):
        self._check_time()
        key = self._position_key(color)
        cached = self._rank_cache.get(key)
        if cached is not None:
            self.last_rank_hits += 1
            ranked, forcing = cached
            self._remember(self._forcing_cache, key, forcing)
            return list(ranked)
        wins, draws, threats = self._tactical_state(color)
        if wins:
            ranked = [(MATE, move) for move in wins]
            self._remember(self._rank_cache, key, (tuple(ranked), ()))
            return ranked
        candidates = self._candidates()
        if threats:
            candidates = self._defending_moves(color) or candidates
        ranked = self._score_moves(candidates, color)
        self._remember(self._rank_cache, key, (tuple(ranked), self._forcing_cache[key]))
        return ranked

    def _score_moves(self, candidates, color):
        opponent = 1 - color
        result, forcing = [], []
        key = self._position_key(color)
        scored = self._move_score_cache.get(key)
        if scored is None:
            scored = {}
            self._remember(self._move_score_cache, key, scored)
        for move in candidates:
            self._check_time()
            cached = scored.get(move)
            if cached is not None:
                self.last_move_score_hits += 1
                score, is_forcing = cached
                result.append((score, move))
                if is_forcing:
                    forcing.append(move)
                continue
            is_forcing = False
            captured = self._make_move(move, color)
            try:
                outcome = self._outcome(move, color, captured)
                local = self._local_score(move, color)
                attack = local + 600 * len(captured)
                attack -= 900 * self._vulnerable_pairs(move, color)
                if outcome == 0:
                    is_forcing = True
                elif outcome is None and (local >= 8_000 or captured):
                    if self._winning_options(color)[0]:
                        is_forcing = True
            finally:
                self._undo_move(move, color, captured)
            captured = self._make_move(move, opponent)
            try:
                defence = self._local_score(move, opponent) + 600 * len(captured)
            finally:
                self._undo_move(move, opponent, captured)
            score = attack + 0.95 * defence - self._centre_distance(move) * 0.1
            if outcome == -1:
                score = -MATE
            elif outcome == 0:
                score = -50
            elif outcome == 1:
                score = MATE
            scored[move] = score, is_forcing
            if is_forcing:
                forcing.append(move)
            result.append((score, move))
        self._remember(self._forcing_cache, key, tuple(forcing))
        return sorted(result, reverse=True)

    def _select_moves(self, ranked, color, width):
        if self._tactical_state(color)[2]:
            return ranked
        forcing = set(self._forcing_cache.get(self._position_key(color), ()))
        selected = list(ranked[:width])
        selected.extend(item for item in ranked[width:] if item[1] in forcing)
        return selected

    def _position_key(self, color):
        self._ensure_position()
        if self._board_key is None:
            self._board_key = bytes(self._state)
        return color, self._board_key

    def _potential_wins(self, color):
        key = self._position_key(color)
        cached = self._potential_cache.get(key)
        if cached is not None:
            return cached
        grid, result = self._grid, set()
        for move in self._stones[color]:
            if grid[move] != color:
                continue
            for axis in AXES:
                if grid[move - axis] == color:
                    continue
                length, end = 0, move
                while grid[end] == color:
                    length += 1
                    end += axis
                if length == 4:
                    before = move - axis
                    if grid[before] == EMPTY and grid[before - axis] != color:
                        result.add(before)
                    if grid[end] == EMPTY and grid[end + axis] != color:
                        result.add(end)
                elif length < 4 and grid[end] == EMPTY:
                    tail, count = end + axis, 0
                    while grid[tail] == color:
                        count += 1
                        tail += axis
                    if length + count == 4:
                        result.add(end)
        result = tuple(sorted(result))
        self._remember(self._potential_cache, key, result)
        return result

    def _winning_options(self, color):
        self._check_time()
        key = self._position_key(color)
        cached = self._winning_cache.get(key)
        if cached is not None:
            self.last_winning_hits += 1
            return cached
        wins, draws = [], []
        for move in self._potential_wins(color):
            self._check_time()
            captured = self._make_move(move, color)
            try:
                outcome = self._outcome(move, color, captured)
                if outcome == 1:
                    wins.append(move)
                elif outcome == 0:
                    draws.append(move)
            finally:
                self._undo_move(move, color, captured)
        result = tuple(wins), tuple(draws)
        self._remember(self._winning_cache, key, result)
        return result

    def _tactical_state(self, color):
        self._check_time()
        key = self._position_key(color)
        if key not in self._tactical_cache:
            wins, draws = self._winning_options(color)
            threats = self._winning_options(1 - color)[0]
            self._remember(self._tactical_cache, key, (wins, draws, threats))
        return self._tactical_cache[key]

    def _defending_moves(self, color):
        key = self._position_key(color)
        if key in self._defence_cache:
            return self._defence_cache[key]
        safe = []
        for move in self._candidates():
            self._check_time()
            captured = self._make_move(move, color)
            try:
                outcome = self._outcome(move, color, captured)
                if outcome in (0, 1):
                    safe.append(move)
                elif outcome is None and not self._winning_options(1 - color)[0]:
                    safe.append(move)
            finally:
                self._undo_move(move, color, captured)
        self._remember(self._defence_cache, key, tuple(safe))
        return self._defence_cache[key]

    def _forcing_candidates(self, color):
        grid, near = self._grid, set()
        for line in self._lines:
            self._check_time()
            for start in range(len(line) - 4):
                cells = line[start:start + 5]
                values = [grid[p] for p in cells]
                if values.count(color) == 3 and values.count(EMPTY) == 2:
                    near.update(p for p in cells if grid[p] == EMPTY)
        for move in CELLS:
            if grid[move] != color:
                continue
            for direction in DIRECTIONS:
                end = move + 3 * direction
                if 0 <= end < len(grid) and grid[end] == EMPTY:
                    if grid[move + direction] == grid[move + 2 * direction] == 1 - color:
                        near.add(end)
        for move in self._potential_wins(color):
            for direction in DIRECTIONS:
                end = move + 3 * direction
                if 0 <= end < len(grid) and grid[end] == EMPTY:
                    if grid[move + direction] == grid[move + 2 * direction] == 1 - color:
                        near.add(end)
        return tuple(sorted(near))

    def _forcing_moves(self, color):
        key = self._position_key(color)
        if key in self._forcing_cache:
            return self._score_moves(self._forcing_cache[key], color)
        forcing = []
        for move in self._forcing_candidates(color):
            self._check_time()
            captured = self._make_move(move, color)
            try:
                outcome = self._outcome(move, color, captured)
                if outcome == 0 or outcome == 1:
                    forcing.append(move)
                elif outcome is None and self._winning_options(color)[0]:
                    forcing.append(move)
            finally:
                self._undo_move(move, color, captured)
        return self._score_moves(forcing, color)

    def _quiescence(self, color, alpha, beta, ply, remaining):
        self._check_time()
        self._ensure_position()
        self.last_qnodes += 1
        self.last_extension = max(self.last_extension, self._q_depth - remaining)
        key = (self._position_key(color), remaining, ply)
        cached = self._q_cache.get(key)
        if cached is not None:
            value, flag = cached
            if flag == 0 or flag == 1 and value >= beta or flag == -1 and value <= alpha:
                self.last_tt_hits += 1
                return value
        value = self._quiescence_uncached(color, alpha, beta, ply, remaining)
        self._remember(self._q_cache, key, (value, self._bound(value, alpha, beta)))
        return value

    def _quiescence_uncached(self, color, alpha, beta, ply, remaining):
        if self._empty_count == 0:
            return 0
        wins, draws, threats = self._tactical_state(color)
        if wins:
            return MATE - ply
        if threats:
            safe = self._defending_moves(color)
            if not safe:
                return -MATE + ply + 1
            if remaining == 0:
                score = self._evaluate(color)
                return max(score, 0) if draws else score
            moves = self._score_moves(safe, color)
            best = -2 * MATE
        else:
            best = self._evaluate(color)
            if draws:
                best = max(best, 0)
            if remaining == 0 or best >= beta:
                return best
            alpha = max(alpha, best)
            moves = self._forcing_moves(color)[:FORCING_WIDTH]
        for _, move in moves:
            self._check_time()
            captured = self._make_move(move, color)
            try:
                outcome = self._outcome(move, color, captured)
                if outcome is not None:
                    score = outcome * (MATE - ply)
                else:
                    score = -self._quiescence(1 - color, -beta, -alpha, ply + 1, remaining - 1)
            finally:
                self._undo_move(move, color, captured)
            best = max(best, score)
            alpha = max(alpha, score)
            if alpha >= beta:
                break
        return best

    def _candidates(self):
        key = self._position_key(0)[1]
        cached = self._candidate_cache.get(key)
        if cached is None:
            near = {q for side in self._stones for p in side for q in self._neighbors[p]
                    if self._grid[q] == EMPTY}
            cached = tuple(sorted(near)) or tuple(p for p in CELLS if self._grid[p] == EMPTY)
            self._remember(self._candidate_cache, key, cached)
        return list(cached)

    def _make_move(self, move, color):
        self._ensure_position()
        previous_key = self._board_key
        self.last_make_calls += 1
        grid = self._grid
        grid[move] = color
        opponent = 1 - color
        captured = []
        for direction in DIRECTIONS:
            a, b, end = move + direction, move + 2 * direction, move + 3 * direction
            if 0 <= end < len(grid) and grid[a] == opponent and grid[b] == opponent and grid[end] == color:
                captured.extend((a, b))
        for p in captured:
            grid[p] = EMPTY
        self._stones[color].add(move)
        self._state[move] = color + 1
        for p in captured:
            self._stones[1 - color].remove(p)
            self._state[p] = 0
        self._empty_count += len(captured) - 1
        self._board_key = None
        self._move_stack.append(previous_key)
        return captured

    def _undo_move(self, move, color, captured):
        self._board_key = self._move_stack.pop()
        self._grid[move] = EMPTY
        self._state[move] = 0
        self._stones[color].remove(move)
        self._empty_count += 1 - len(captured)
        for p in captured:
            self._grid[p] = 1 - color
            self._state[p] = 2 - color
            self._stones[1 - color].add(p)

    def _exact_five_at(self, move, color):
        if self._grid[move] != color:
            return False
        for axis in AXES:
            length = 1
            for direction in (axis, -axis):
                p = move + direction
                while 0 <= p < len(self._grid) and self._grid[p] == color:
                    length += 1
                    p += direction
            if length == 5:
                return True
        return False

    def _outcome(self, move, color, captured):
        mine = self._exact_five_at(move, color)
        opponent = 1 - color
        theirs = False
        for removed in captured:
            for direction in DIRECTIONS:
                p = removed + direction
                if 0 <= p < len(self._grid) and self._grid[p] == opponent:
                    if self._exact_five_at(p, opponent):
                        theirs = True
                        break
            if theirs:
                break
        if mine and theirs:
            return 0
        if mine:
            return 1
        if theirs:
            return -1
        return None

    def _local_score(self, move, color):
        grid = self._grid
        total = 0
        for axis in AXES:
            length, open_ends = 1, 0
            for direction in (axis, -axis):
                p = move + direction
                while 0 <= p < len(grid) and grid[p] == color:
                    length += 1
                    p += direction
                if 0 <= p < len(grid) and grid[p] == EMPTY:
                    open_ends += 1
            if length == 4:
                total += 45_000 if open_ends == 2 else 6_000 if open_ends else 0
            elif length == 3:
                total += 3_000 if open_ends == 2 else 250 if open_ends else 0
            elif length == 2:
                total += 120 if open_ends == 2 else 20 if open_ends else 0
            for offset in range(5):
                start = move - offset * axis
                end = start + 4 * axis
                if not (0 <= start < len(grid) and 0 <= end < len(grid)):
                    continue
                cells = [grid[start + i * axis] for i in range(5)]
                if any(value not in (EMPTY, color) for value in cells):
                    continue
                before, after = start - axis, end + axis
                if 0 <= before < len(grid) and grid[before] == color:
                    continue
                if 0 <= after < len(grid) and grid[after] == color:
                    continue
                total += WINDOW_WEIGHTS[cells.count(color)]
        return total

    def _vulnerable_pairs(self, move, color):
        count, grid = 0, self._grid
        for direction in DIRECTIONS:
            a, before, after = move + direction, move - direction, move + 2 * direction
            if not (0 <= before < len(grid) and 0 <= after < len(grid)):
                continue
            if grid[a] == color and {grid[before], grid[after]} == {1 - color, EMPTY}:
                count += 1
        return count

    def _evaluate(self, color):
        key = self._position_key(0)[1]
        total = self._eval_cache.get(key)
        if total is None:
            total = 0
            for line in self._lines:
                values = tuple(self._grid[p] for p in line)
                score = self._line_cache.get(values)
                if score is None:
                    score = self._score_line(values, 0) - self._score_line(values, 1)
                    self._remember(self._line_cache, values, score, LINE_CACHE_LIMIT)
                else:
                    self.last_line_hits += 1
                total += score
            self._remember(self._eval_cache, key, total)
        return total if color == 0 else -total

    @staticmethod
    def _bound(value, alpha, beta):
        return -1 if value <= alpha else 1 if value >= beta else 0

    @staticmethod
    def _remember(cache, key, value, limit=POSITION_CACHE_LIMIT):
        if key not in cache and len(cache) >= limit:
            if isinstance(cache, OrderedDict):
                cache.popitem(last=False)
            else:
                cache.pop(next(iter(cache)))
        cache[key] = value

    def _ensure_position(self):
        if self._indexed_grid is self._grid:
            return
        self._indexed_grid = self._grid
        self._state = bytearray(value + 1 for value in self._grid)
        self._board_key = None
        self._stones = [{p for p in CELLS if self._grid[p] == color} for color in (0, 1)]
        self._empty_count = SIZE * SIZE - sum(map(len, self._stones))
        self._move_stack = []
        for name in ('_tactical_cache', '_defence_cache', '_forcing_cache',
                     '_winning_cache', '_potential_cache', '_rank_cache', '_move_score_cache',
                     '_candidate_cache', '_search_cache', '_q_cache', '_eval_cache'):
            setattr(self, name, OrderedDict())




    def _score_line(self, values, color):
        total = 0
        for start in range(len(values) - 4):
            window = values[start:start + 5]
            if any(value not in (EMPTY, color) for value in window):
                continue
            if start and values[start - 1] == color:
                continue
            if start + 5 < len(values) and values[start + 5] == color:
                continue
            count = window.count(color)
            total += WINDOW_WEIGHTS[count]
        i = 0
        while i < len(values):
            if values[i] != color:
                i += 1
                continue
            start = i
            while i < len(values) and values[i] == color:
                i += 1
            length = i - start
            ends = int(start > 0 and values[start - 1] == EMPTY) + int(i < len(values) and values[i] == EMPTY)
            if length == 4 and ends == 2:
                total += 30_000
            elif length == 3 and ends == 2:
                total += 2_000
            elif length == 2 and ends == 2:
                total += 80
        for i in range(len(values) - 3):
            pair = values[i:i + 4]
            if pair in ((1 - color, color, color, EMPTY), (EMPTY, color, color, 1 - color)):
                total -= 700
        return total

    def _build_lines(self):
        lines = []
        for axis in AXES:
            for p in CELLS:
                r, c = self._coordinates(p - axis)
                if 0 <= r < SIZE and 0 <= c < SIZE:
                    continue
                line, q = [], p
                while True:
                    r, c = self._coordinates(q)
                    if not (0 <= r < SIZE and 0 <= c < SIZE):
                        break
                    line.append(q)
                    q += axis
                if len(line) >= 4:
                    lines.append(tuple(line))
        return tuple(lines)

    def _coordinates(self, move):
        return move // STRIDE - 1, move % STRIDE - 1

    def _centre_distance(self, move):
        r, c = self._coordinates(move)
        return abs(r - SIZE // 2) + abs(c - SIZE // 2)

    def _check_time(self):
        if perf_counter() >= self._deadline:
            raise TimeoutError

팀명: JangJinha
팀원: 장진하 (개인 참가)

이번 변경
첫 탐색 단계에서는 전술 연장 탐색을 0수로 두어 후보 전체를 가볍게 비교합니다.
즉시 승리, 상대의 즉시 승리 위협, 포획으로 방어할 수 있는지는 계속 검사합니다.
두 번째 탐색 단계부터는 기존처럼 최대 6수의 전술 연장 탐색을 사용합니다.
일반 탐색의 재사용 기록도 연장 길이를 구분하여 얕은 결과가 깊은 결과를
대신하지 않게 했습니다. 차례가 끝나거나 탐색이 중단되면 설정을 복구합니다.

설정
평가 점수·가중치, 후보 범위 2칸, 첫 후보 20개/이후 12개,
일반 탐색 최대 6수, 두 번째 단계 이후 전술 연장 최대 6수,
공격 후보 최대 8개, 필수 방어 수를 모두 검토하는 방식은 v3와 같습니다.
포획, 정확히 5목, 동시 승리 무승부 규칙과 v3의 계산 재사용도 같습니다.
한 차례 시간 예산은 min(1초, 남은 게임 시간/30, 남은 시간-0.02초)입니다.
시간 제한이 없을 때도 내부 예산은 한 차례 최대 1초입니다.

실행 방법
Python 3 표준 라이브러리만 사용합니다.
ZIP을 풀고 agent_JangJinha_v4 폴더에서 아래 명령을 실행합니다.
기존 board.py, player.py, humanplayer.py, randomplayer.py, simpleplayer.py가
있는 폴더의 경로를 지정합니다.

사람(흑) 대 JangJinha(백), 게임 전체 시간 제한 없음:
python3 run_agent.py "/기존/오목/폴더" 0 3 -1
JangJinha(흑) 대 SimplePlayer(백), 각자 게임 전체 누적 1분:
python3 run_agent.py "/기존/오목/폴더" 3 2 1

흑=0, 백=1. 보드 빈칸=-1. 좌표는 0부터 세는 (행, 열) 튜플입니다.
클래스명·모듈명은 v3와 같습니다. 두 에이전트 파일을 함께 사용합니다.

이번 검증 결과
규칙 모의 실행 1170회, 정확히 5목 후보 비교 200회 통과.
착수·포획·되돌리기 등 800회 비교와 동일 깊이 탐색 값 8개 조건 통과.
연장 길이에 따른 재사용 구분, 첫 단계 20개 전체 비교,
두 번째 단계 연장 복원, 시간 종료·예외 발생 후 판 복구 검사 통과.
같은 1초의 24회 판면 비교에서 첫 단계 전체 비교 완료: v3 22회 -> v4 24회.
문제 장면(3번 경기 23수 직전)은 v3가 12~13개에서 중단된 반면,
v4는 20개 전체를 비교하고 (9, 4)를 선택했습니다.
평균 일반 탐색 완료 깊이는 두 버전 모두 1.67수였습니다.
24회 중 v4가 더 깊은 경우 2회, 같은 경우 18회, 더 얕은 경우 4회였습니다.
이 수치는 판면 재탐색 결과입니다. v4의 새 대전 승률은 측정하지 않았습니다.

범위와 한계
여기서 후보 전체는 기존에 선택한 후보 목록입니다. 일반적으로 첫 20개에
추가 공격 후보가 포함되며, 즉시 패배 위협이 있으면 필수 방어 수를 검토합니다.
모든 361칸을 검색하거나 첫 단계부터 6수 연장을 모두 하는 뜻은 아닙니다.
즉시 승리 또는 강제 승리를 확인한 경우에는 기존처럼 조기 반환할 수 있습니다.
남은 시간이 매우 짧거나 전술 검사 비용이 큰 판에서는 첫 단계도 중단될 수
있으며, 시간 제한을 지키기 위해 그때까지 얻은 합법적인 후보를 반환합니다.
(9, 4)의 선택이 경기 전체의 승리를 보장하는 것은 아닙니다.

# JangJinha v4 변경 설명

첫 번째 탐색은 선택된 후보를 끝까지 비교하는 데 집중하고, 두 번째 탐색부터
기존의 깊은 전술 검토를 하는 방식으로 바꿨습니다. 평가 가중치나 후보 폭,
차례당 시간 예산은 v3와 같습니다.

## 바꾼 이유

v3는 첫 번째 후보부터 최대 6수의 전술 연장을 했습니다. 앞쪽 후보 몇 개에
시간을 많이 쓰면 뒤쪽 후보를 보기도 전에 1초가 끝날 수 있었습니다.
이전 10판의 3번 경기 23수 직전에는 중요한 (9, 4)가 후보 순위 16번이었고,
기존 진단에서 그 후보에 도달하지 못했습니다.

## 작동 방식

1. 첫 단계는 연장 길이를 0으로 둡니다. 후보 착수 후 즉시 승리와 상대의 즉시
   승리 위협을 검사합니다. 위협이 있으면 포획 등을 통해 막을 수 있는지도
   확인하고, 더 긴 전술 수순은 펼치지 않은 채 판 평가로 비교합니다.
2. 후보 전체 비교를 마친 뒤, 다음 단계에서는 기존 최대 6수 연장 탐색으로
   들어갑니다. 시간이 끝나면 마지막으로 완료한 단계의 선택을 사용합니다.
3. 일반 탐색 기록에는 판, 색상, 일반 깊이, 수순 거리와 함께 전술 연장 길이도
   넣었습니다. 첫 단계의 얕은 결과가 다음 단계의 깊은 결과를 대신할 수 없습니다.
   전술 연장 기록은 이미 남은 연장 길이를 구분하므로 기존 방식을 유지합니다.
4. 차례 종료나 시간 종료, 예외 발생 시 연장 설정을 기본값 6으로 되돌립니다.
   탐색 중 임시로 둔 돌과 포획한 돌도 기존 방식대로 복구합니다.

## 같은 판면에서 비교한 결과

이전 대전에서 저장한 12개 판면을 각각 두 번씩, 버전별 1초 예산으로 검색했습니다.
매번 새 인스턴스로 시작해 줄 평가 캐시도 비어 있으며, 순서 효과를 줄이기 위해
v3와 v4의 실행 순서를 번갈아 바꿨습니다. 두 버전에 같은 소량의 기록 기능을
붙였고, 검사 중 판의 변경과 불법 착수가 없는지도 확인했습니다.

| 항목 | v3 | v4 |
|---|---:|---:|
| 첫 단계 후보 전체 비교 완료 | 22/24 | 24/24 |
| 첫 단계 미완료 | 2회 | 0회 |
| 평균 일반 탐색 완료 깊이 | 1.67수 | 1.67수 |
| 평균 소요 시간 | 1.000066초 | 1.000075초 |

v4의 완료 깊이가 더 깊은 경우는 2회, 같은 경우는 18회,
더 얕은 경우는 4회였습니다. 첫 단계의 수 평가가 달라지면 다음 단계의
후보 순서와 가지치기도 달라지므로 모든 판에서 깊이가 늘어나는 것은 아닙니다.
이번 수정으로 확인한 성과는 후보 비교의 완료율입니다.

문제 장면에서는 다음과 같았습니다. 좌표는 코드와 동일한 0부터 세는 (행, 열)입니다.

| 반복 | v3 비교 완료 후보 | v3 선택/완료 깊이 | v4 비교 완료 후보 | v4 선택/완료 깊이 |
|---|---:|---|---:|---|
| 1 | 13/20 | (11, 9) / 0수 | 20/20 | (9, 4) / 2수 |
| 2 | 12/20 | (11, 9) / 0수 | 20/20 | (9, 4) / 2수 |

(9, 4)는 상대 돌 두 개를 포획하고 다음 수 승리 위협을 만들 수 있는 후보입니다.
이전 분석에서 확인한 그 후보를 이제 실제 제출 코드가 살펴보고 골랐습니다.
이 판면에서 다른 수를 선택했다는 사실은 경기 전체의 승리를 입증하지 않습니다.

## 검증

- 규칙 모의 실행 1170회와 정확히 5목 후보 비교 200회 통과.
- v2 기준 착수·포획·되돌리기 등 800회 비교, 후보 순서 4개 판면,
  같은 깊이의 검색 값 8개 조건에서 일치.
- 즉시 승리·막을 수 없는 위협·포획 방어·동시 승리 무승부 확인.
- 얕은 검색 결과가 깊은 검색 결과와 섞이지 않는지 검사.
- 첫 단계 20개 전체 비교 후 두 번째 단계에서 연장 길이가 복원되는지 검사.
- 시간 종료와 예상하지 못한 예외 발생 후 판 및 탐색 설정 복구 확인.
- v3 원본이 실제 대전 코드와 같은지, 상수·가중치·Player 인터페이스와 나머지
  메서드가 유지되는지 확인.

세부 측정과 검증 결과는 verification 폴더의 JSON 파일에 있습니다.
기존 v3 결과와 코드는 verification/reference_v3에 보관했습니다.

## 적용 범위

전체 비교 대상은 기존에 선택한 첫 후보 목록입니다. 기본 20개에 추가 공격
후보가 포함될 수 있으며, 즉시 패배 위협이 있으면 필수 방어 수를 검토합니다.
시간이 매우 부족하거나 전술 검사 비용이 큰 경우에는 첫 단계도 완료하지 못할
수 있습니다. 기존 시간 제한과 즉시·강제 승리의 조기 반환은 유지합니다.
v4의 새 대전 승률은 아직 측정하지 않았습니다.

# AI usage: ChatGPT (Codex) wrote regression checks for the v4 first pass.
import sys
from pathlib import Path

ROOT = Path(__file__).resolve().parent
sys.path.insert(0, str(ROOT / 'omok-agent'))
from smartplayerJangJinha import SmartPlayerJangJinha, MATE, QUIESCENCE_DEPTH
from test_agent import grid, set_position, oracle


def prepared(board, color=0, extension=0):
    agent = SmartPlayerJangJinha(color)
    set_position(agent, board)
    agent._q_depth = extension
    return agent


def main():
    # Zero extension still detects both sides' immediate wins.
    board = grid()
    for col in range(5, 9):
        board[9][col] = 0
    for color, expected in ((0, MATE - 1), (1, -MATE + 2)):
        agent = prepared(board, color)
        assert agent._search(color, 0, -2 * MATE, 2 * MATE, 1) == expected
        assert agent.last_extension == 0

    # A capture that removes an open four remains a legal defence at zero depth.
    board[8][6], board[10][6] = 0, 1
    agent = prepared(board, 1)
    assert agent._search(1, 0, -2 * MATE, 2 * MATE, 1) > -MATE + 1000
    safe = {agent._coordinates(p) for p in agent._defending_moves(1)}
    assert (7, 6) in safe and (9, 4) not in safe and (9, 9) not in safe
    assert agent._defence_cache and agent.last_extension == 0

    # Captures can reveal an opposing five: simultaneous wins stay draws.
    board = grid()
    for col in range(5, 11):
        board[8][col] = 1
    board[9][5], board[10][5] = 1, 0
    for col in range(1, 5):
        board[7][col] = 0
    assert oracle(board, (7, 5), 0)[2] == 0
    agent = prepared(board)
    wins, draws = agent._winning_options(0)
    draw = (7 + 1) * 21 + 5 + 1
    assert draw in draws and draw not in wins
    taken = agent._make_move(draw, 0)
    try:
        assert agent._outcome(draw, 0, taken) == 0
    finally:
        agent._undo_move(draw, 0, taken)

    # A shallow exact value cannot suppress a deeper forced-win search.
    board = grid()
    for col in (8, 9, 10):
        board[9][col] = 0
    agent = prepared(board)
    key = agent._position_key(0)
    shallow = agent._search(0, 0, -2 * MATE, 2 * MATE, 1)
    assert shallow < MATE - 1000 and agent.last_extension == 0
    agent._q_depth = QUIESCENCE_DEPTH
    deep = agent._search(0, 0, -2 * MATE, 2 * MATE, 1)
    assert deep > MATE - 100 and deep != shallow
    assert (key, 0, 1, 0) in agent._search_cache
    assert (key, 0, 1, QUIESCENCE_DEPTH) in agent._search_cache
    for extension, expected in ((0, shallow), (QUIESCENCE_DEPTH, deep)):
        agent._q_depth = extension
        hits = agent.last_tt_hits
        assert agent._search(0, 0, -2 * MATE, 2 * MATE, 1) == expected
        assert agent.last_tt_hits > hits

    # Stop at the second iteration boundary and inspect the complete first pass.
    board = grid()
    board[9][9], board[8][9], board[10][10] = 0, 1, 0
    original = [row[:] for row in board]
    agent = SmartPlayerJangJinha(0)
    select, search = agent._select_moves, agent._search
    modes, completed, expected_count = [], [], []
    def inspect_selection(ranked, color, width):
        selected = select(ranked, color, width)
        if not agent._move_stack:
            modes.append(agent._q_depth)
            if len(modes) == 2:
                raise TimeoutError
            expected_count.append(len(selected))
        return selected
    def inspect_search(color, depth, alpha, beta, ply):
        value = search(color, depth, alpha, beta, ply)
        if ply == 1:
            completed.append((depth, agent._q_depth))
        return value
    agent._select_moves, agent._search = inspect_selection, inspect_search
    move = agent.take_turn(board, 60_000)
    assert modes == [0, QUIESCENCE_DEPTH]
    assert len(completed) == expected_count[0] == 20
    assert all(item == (0, 0) for item in completed)
    assert agent.last_depth == 1 and agent.last_extension == 0
    assert agent._q_depth == QUIESCENCE_DEPTH
    assert board == original and board[move[0]][move[1]] == -1
    assert not agent._move_stack

    # Cancellation or an unexpected error inside a root move restores the mode.
    for exception in (TimeoutError, RuntimeError):
        agent = SmartPlayerJangJinha(0)
        def stop(*args):
            assert len(agent._move_stack) == 1 and agent._q_depth == 0
            raise exception('injected during first candidate')
        agent._search = stop
        try:
            move = agent.take_turn(board, 60_000)
        except RuntimeError:
            assert exception is RuntimeError
        else:
            assert exception is TimeoutError and board[move[0]][move[1]] == -1
        assert agent._q_depth == QUIESCENCE_DEPTH

def traced(base):
    class Trace(base):
        def __init__(self, color):
            super().__init__(color)
            self.root_iteration = 0
            self.first_expected = 0
            self.first_trace = []
            self.first_complete_at = None
            self.root_move = None

        def _select_moves(self, ranked, color, width):
            selected = super()._select_moves(ranked, color, width)
            if not self._move_stack:
                self.root_iteration += 1
                if self.root_iteration == 1:
                    self.first_expected = len(selected)
            return selected

        def _make_move(self, move, color):
            captured = super()._make_move(move, color)
            if len(self._move_stack) == 1:
                self.root_move = move
            return captured

        def record(self, move, started, score=None, completed=False):
            self.first_trace.append({'move': list(self._coordinates(move)),
                                     'completed': completed,
                                     'seconds': perf_counter() - started,
                                     'score': score})
            if sum(row['completed'] for row in self.first_trace) == self.first_expected:
                self.first_complete_at = perf_counter()

        def _outcome(self, move, color, captured):
            value = super()._outcome(move, color, captured)
            if self.root_iteration == 1 and len(self._move_stack) == 1 and value is not None:
                self.record(move, perf_counter(), value * 10_000_000, True)
            return value

        def _search(self, color, depth, alpha, beta, ply):
            if self.root_iteration != 1 or ply != 1:
                return super()._search(color, depth, alpha, beta, ply)
            move, started = self.root_move, perf_counter()
            try:
                value = super()._search(color, depth, alpha, beta, ply)
            except TimeoutError:
                self.record(move, started)
                raise
            self.record(move, started, -value, True)
            return value
    return Trace


def write(path, data):
    temporary = path.with_suffix('.tmp')
    temporary.write_text(json.dumps(data, ensure_ascii=False, indent=2), encoding='utf-8')
    temporary.replace(path)
