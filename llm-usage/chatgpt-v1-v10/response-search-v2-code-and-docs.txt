from time import perf_counter

from playerTeamName import PlayerTeamName

SIZE = 19
EMPTY = -1
WALL = 2
STRIDE = SIZE + 2
AXES = (1, STRIDE, STRIDE + 1, STRIDE - 1)
DIRECTIONS = AXES + tuple(-d for d in AXES)
CELLS = tuple((r + 1) * STRIDE + c + 1 for r in range(SIZE) for c in range(SIZE))
MATE = 10_000_000
WINDOW_WEIGHTS = (0, 2, 30, 450, 8_000, 0)
MAX_DEPTH = 6
QUIESCENCE_DEPTH = 6
ROOT_WIDTH = 20
INNER_WIDTH = 12
FORCING_WIDTH = 8


class SmartPlayerTeamName(PlayerTeamName):
    def __init__(self, color):
        super().__init__(color)
        if color not in (0, 1):
            raise ValueError("color must be 0 (black) or 1 (white)")
        self._lines = self._build_lines()
        self.last_depth = 0
        self.last_nodes = 0
        self.last_qnodes = 0
        self.last_extension = 0
        self._tactical_cache = {}
        self._defence_cache = {}
        self._forcing_cache = {}

    def take_turn(self, board, time):
        started = perf_counter()
        self._grid = [WALL] * (STRIDE * STRIDE)
        for r in range(SIZE):
            for c in range(SIZE):
                self._grid[(r + 1) * STRIDE + c + 1] = int(board[r][c])
        empty = [p for p in CELLS if self._grid[p] == EMPTY]
        if not empty:
            raise ValueError("No legal move: the board is full.")
        if len(empty) == SIZE * SIZE:
            return SIZE // 2, SIZE // 2

        candidates = self._candidates()
        best = min(candidates, key=self._centre_distance)
        remaining = float(time) / 1000 if time != -1 else None
        if remaining is not None and remaining <= 0.015:
            return self._coordinates(best)
        budget = 1.0 if remaining is None else min(1.0, remaining / 30, max(0, remaining - 0.02))
        self._deadline = started + budget
        self._line_cache = {}
        self._tactical_cache = {}
        self._defence_cache = {}
        self._forcing_cache = {}
        self.last_depth = 0
        self.last_nodes = 0
        self.last_qnodes = 0
        self.last_extension = 0

        try:
            ranked = self._rank_moves(self.color)
            if ranked:
                best = ranked[0][1]
            if ranked and ranked[0][0] >= MATE:
                return self._coordinates(best)
            for depth in range(1, MAX_DEPTH + 1):
                selected = self._select_moves(ranked, self.color, ROOT_WIDTH)
                ordered = sorted(selected, key=lambda item: item[1] != best)
                depth_best, depth_score = best, -2 * MATE
                alpha = -2 * MATE
                for _, move in ordered:
                    self._check_time()
                    captured = self._make_move(move, self.color)
                    try:
                        outcome = self._outcome(move, self.color, captured)
                        if outcome is not None:
                            score = outcome * MATE
                        else:
                            score = -self._search(1 - self.color, depth - 1, -2 * MATE, -alpha, 1)
                    finally:
                        self._undo_move(move, self.color, captured)
                    if score > depth_score:
                        depth_best, depth_score = move, score
                        if depth == 1:
                            best = depth_best
                    alpha = max(alpha, score)
                    if depth == 1 and score > MATE - 1000:
                        self.last_depth = 1
                        return self._coordinates(move)
                best = depth_best
                self.last_depth = depth
        except TimeoutError:
            pass
        return self._coordinates(best)

    def _search(self, color, depth, alpha, beta, ply):
        self._check_time()
        self.last_nodes += 1
        if not any(self._grid[p] == EMPTY for p in CELLS):
            return 0
        if depth == 0:
            return self._quiescence(color, alpha, beta, ply, QUIESCENCE_DEPTH)
        wins, draws, threats = self._tactical_state(color)
        if wins:
            return MATE - ply
        if threats and not self._defending_moves(color):
            return -MATE + ply + 1
        moves = self._rank_moves(color)
        if moves[0][0] >= MATE:
            return MATE - ply
        best = -2 * MATE
        for _, move in self._select_moves(moves, color, INNER_WIDTH):
            self._check_time()
            captured = self._make_move(move, color)
            try:
                outcome = self._outcome(move, color, captured)
                if outcome is not None:
                    score = outcome * (MATE - ply)
                else:
                    score = -self._search(1 - color, depth - 1, -beta, -alpha, ply + 1)
            finally:
                self._undo_move(move, color, captured)
            best = max(best, score)
            alpha = max(alpha, score)
            if alpha >= beta:
                break
        return best

    def _rank_moves(self, color):
        wins, draws, threats = self._tactical_state(color)
        if wins:
            return [(MATE, move) for move in wins]
        candidates = self._candidates()
        if threats:
            candidates = self._defending_moves(color) or candidates
        return self._score_moves(candidates, color)

    def _score_moves(self, candidates, color):
        opponent = 1 - color
        result, forcing = [], []
        key = self._position_key(color)
        for move in candidates:
            self._check_time()
            captured = self._make_move(move, color)
            try:
                outcome = self._outcome(move, color, captured)
                local = self._local_score(move, color)
                attack = local + 600 * len(captured)
                attack -= 900 * self._vulnerable_pairs(move, color)
                if outcome == 0:
                    forcing.append(move)
                elif outcome is None and (local >= 8_000 or captured):
                    if self._winning_options(color)[0]:
                        forcing.append(move)
            finally:
                self._undo_move(move, color, captured)
            captured = self._make_move(move, opponent)
            try:
                defence = self._local_score(move, opponent) + 600 * len(captured)
            finally:
                self._undo_move(move, opponent, captured)
            score = attack + 0.95 * defence - self._centre_distance(move) * 0.1
            if outcome == -1:
                score = -MATE
            elif outcome == 0:
                score = -50
            elif outcome == 1:
                score = MATE
            result.append((score, move))
        self._forcing_cache[key] = tuple(forcing)
        return sorted(result, reverse=True)

    def _select_moves(self, ranked, color, width):
        if self._tactical_state(color)[2]:
            return ranked
        forcing = set(self._forcing_cache.get(self._position_key(color), ()))
        selected = list(ranked[:width])
        selected.extend(item for item in ranked[width:] if item[1] in forcing)
        return selected

    def _position_key(self, color):
        return color, tuple(self._grid)

    def _potential_wins(self, color):
        grid, result = self._grid, set()
        for move in CELLS:
            if grid[move] != color:
                continue
            for axis in AXES:
                if grid[move - axis] == color:
                    continue
                length, end = 0, move
                while grid[end] == color:
                    length += 1
                    end += axis
                if length == 4:
                    before = move - axis
                    if grid[before] == EMPTY and grid[before - axis] != color:
                        result.add(before)
                    if grid[end] == EMPTY and grid[end + axis] != color:
                        result.add(end)
                elif length < 4 and grid[end] == EMPTY:
                    tail, count = end + axis, 0
                    while grid[tail] == color:
                        count += 1
                        tail += axis
                    if length + count == 4:
                        result.add(end)
        return tuple(sorted(result))

    def _winning_options(self, color):
        wins, draws = [], []
        for move in self._potential_wins(color):
            self._check_time()
            captured = self._make_move(move, color)
            try:
                outcome = self._outcome(move, color, captured)
                if outcome == 1:
                    wins.append(move)
                elif outcome == 0:
                    draws.append(move)
            finally:
                self._undo_move(move, color, captured)
        return tuple(wins), tuple(draws)

    def _tactical_state(self, color):
        self._check_time()
        key = self._position_key(color)
        if key not in self._tactical_cache:
            wins, draws = self._winning_options(color)
            threats = self._winning_options(1 - color)[0]
            self._tactical_cache[key] = wins, draws, threats
        return self._tactical_cache[key]

    def _defending_moves(self, color):
        key = self._position_key(color)
        if key in self._defence_cache:
            return self._defence_cache[key]
        safe = []
        for move in self._candidates():
            self._check_time()
            captured = self._make_move(move, color)
            try:
                outcome = self._outcome(move, color, captured)
                if outcome in (0, 1):
                    safe.append(move)
                elif outcome is None and not self._winning_options(1 - color)[0]:
                    safe.append(move)
            finally:
                self._undo_move(move, color, captured)
        self._defence_cache[key] = tuple(safe)
        return self._defence_cache[key]

    def _forcing_candidates(self, color):
        grid, near = self._grid, set()
        for line in self._lines:
            self._check_time()
            for start in range(len(line) - 4):
                cells = line[start:start + 5]
                values = [grid[p] for p in cells]
                if values.count(color) == 3 and values.count(EMPTY) == 2:
                    near.update(p for p in cells if grid[p] == EMPTY)
        for move in CELLS:
            if grid[move] != color:
                continue
            for direction in DIRECTIONS:
                end = move + 3 * direction
                if 0 <= end < len(grid) and grid[end] == EMPTY:
                    if grid[move + direction] == grid[move + 2 * direction] == 1 - color:
                        near.add(end)
        for move in self._potential_wins(color):
            for direction in DIRECTIONS:
                end = move + 3 * direction
                if 0 <= end < len(grid) and grid[end] == EMPTY:
                    if grid[move + direction] == grid[move + 2 * direction] == 1 - color:
                        near.add(end)
        return tuple(sorted(near))

    def _forcing_moves(self, color):
        key = self._position_key(color)
        if key in self._forcing_cache:
            return self._score_moves(self._forcing_cache[key], color)
        forcing = []
        for move in self._forcing_candidates(color):
            self._check_time()
            captured = self._make_move(move, color)
            try:
                outcome = self._outcome(move, color, captured)
                if outcome == 0 or outcome == 1:
                    forcing.append(move)
                elif outcome is None and self._winning_options(color)[0]:
                    forcing.append(move)
            finally:
                self._undo_move(move, color, captured)
        return self._score_moves(forcing, color)

    def _quiescence(self, color, alpha, beta, ply, remaining):
        self._check_time()
        self.last_qnodes += 1
        self.last_extension = max(self.last_extension, QUIESCENCE_DEPTH - remaining)
        if not any(self._grid[p] == EMPTY for p in CELLS):
            return 0
        wins, draws, threats = self._tactical_state(color)
        if wins:
            return MATE - ply
        if threats:
            safe = self._defending_moves(color)
            if not safe:
                return -MATE + ply + 1
            if remaining == 0:
                score = self._evaluate(color)
                return max(score, 0) if draws else score
            moves = self._score_moves(safe, color)
            best = -2 * MATE
        else:
            best = self._evaluate(color)
            if draws:
                best = max(best, 0)
            if remaining == 0 or best >= beta:
                return best
            alpha = max(alpha, best)
            moves = self._forcing_moves(color)[:FORCING_WIDTH]
        for _, move in moves:
            self._check_time()
            captured = self._make_move(move, color)
            try:
                outcome = self._outcome(move, color, captured)
                if outcome is not None:
                    score = outcome * (MATE - ply)
                else:
                    score = -self._quiescence(1 - color, -beta, -alpha, ply + 1, remaining - 1)
            finally:
                self._undo_move(move, color, captured)
            best = max(best, score)
            alpha = max(alpha, score)
            if alpha >= beta:
                break
        return best

    def _candidates(self):
        grid = self._grid
        near = set()
        for p in CELLS:
            if grid[p] in (0, 1):
                r, c = self._coordinates(p)
                for rr in range(max(0, r - 2), min(SIZE, r + 3)):
                    for cc in range(max(0, c - 2), min(SIZE, c + 3)):
                        q = (rr + 1) * STRIDE + cc + 1
                        if grid[q] == EMPTY:
                            near.add(q)
        return sorted(near) or [p for p in CELLS if grid[p] == EMPTY]

    def _make_move(self, move, color):
        grid = self._grid
        grid[move] = color
        opponent = 1 - color
        captured = []
        for direction in DIRECTIONS:
            a, b, end = move + direction, move + 2 * direction, move + 3 * direction
            if 0 <= end < len(grid) and grid[a] == opponent and grid[b] == opponent and grid[end] == color:
                captured.extend((a, b))
        for p in captured:
            grid[p] = EMPTY
        return captured

    def _undo_move(self, move, color, captured):
        self._grid[move] = EMPTY
        for p in captured:
            self._grid[p] = 1 - color

    def _exact_five_at(self, move, color):
        if self._grid[move] != color:
            return False
        for axis in AXES:
            length = 1
            for direction in (axis, -axis):
                p = move + direction
                while 0 <= p < len(self._grid) and self._grid[p] == color:
                    length += 1
                    p += direction
            if length == 5:
                return True
        return False

    def _outcome(self, move, color, captured):
        mine = self._exact_five_at(move, color)
        opponent = 1 - color
        theirs = False
        for removed in captured:
            for direction in DIRECTIONS:
                p = removed + direction
                if 0 <= p < len(self._grid) and self._grid[p] == opponent:
                    if self._exact_five_at(p, opponent):
                        theirs = True
                        break
            if theirs:
                break
        if mine and theirs:
            return 0
        if mine:
            return 1
        if theirs:
            return -1
        return None

    def _local_score(self, move, color):
        grid = self._grid
        total = 0
        for axis in AXES:
            length, open_ends = 1, 0
            for direction in (axis, -axis):
                p = move + direction
                while 0 <= p < len(grid) and grid[p] == color:
                    length += 1
                    p += direction
                if 0 <= p < len(grid) and grid[p] == EMPTY:
                    open_ends += 1
            if length == 4:
                total += 45_000 if open_ends == 2 else 6_000 if open_ends else 0
            elif length == 3:
                total += 3_000 if open_ends == 2 else 250 if open_ends else 0
            elif length == 2:
                total += 120 if open_ends == 2 else 20 if open_ends else 0
            for offset in range(5):
                start = move - offset * axis
                end = start + 4 * axis
                if not (0 <= start < len(grid) and 0 <= end < len(grid)):
                    continue
                cells = [grid[start + i * axis] for i in range(5)]
                if any(value not in (EMPTY, color) for value in cells):
                    continue
                before, after = start - axis, end + axis
                if 0 <= before < len(grid) and grid[before] == color:
                    continue
                if 0 <= after < len(grid) and grid[after] == color:
                    continue
                total += WINDOW_WEIGHTS[cells.count(color)]
        return total

    def _vulnerable_pairs(self, move, color):
        count, grid = 0, self._grid
        for direction in DIRECTIONS:
            a, before, after = move + direction, move - direction, move + 2 * direction
            if not (0 <= before < len(grid) and 0 <= after < len(grid)):
                continue
            if grid[a] == color and {grid[before], grid[after]} == {1 - color, EMPTY}:
                count += 1
        return count

    def _evaluate(self, color):
        total = 0
        for line in self._lines:
            values = tuple(self._grid[p] for p in line)
            if values not in self._line_cache:
                self._line_cache[values] = self._score_line(values, 0) - self._score_line(values, 1)
            total += self._line_cache[values]
        return total if color == 0 else -total

    def _score_line(self, values, color):
        total = 0
        for start in range(len(values) - 4):
            window = values[start:start + 5]
            if any(value not in (EMPTY, color) for value in window):
                continue
            if start and values[start - 1] == color:
                continue
            if start + 5 < len(values) and values[start + 5] == color:
                continue
            count = window.count(color)
            total += WINDOW_WEIGHTS[count]
        i = 0
        while i < len(values):
            if values[i] != color:
                i += 1
                continue
            start = i
            while i < len(values) and values[i] == color:
                i += 1
            length = i - start
            ends = int(start > 0 and values[start - 1] == EMPTY) + int(i < len(values) and values[i] == EMPTY)
            if length == 4 and ends == 2:
                total += 30_000
            elif length == 3 and ends == 2:
                total += 2_000
            elif length == 2 and ends == 2:
                total += 80
        for i in range(len(values) - 3):
            pair = values[i:i + 4]
            if pair in ((1 - color, color, color, EMPTY), (EMPTY, color, color, 1 - color)):
                total -= 700
        return total

    def _build_lines(self):
        lines = []
        for axis in AXES:
            for p in CELLS:
                r, c = self._coordinates(p - axis)
                if 0 <= r < SIZE and 0 <= c < SIZE:
                    continue
                line, q = [], p
                while True:
                    r, c = self._coordinates(q)
                    if not (0 <= r < SIZE and 0 <= c < SIZE):
                        break
                    line.append(q)
                    q += axis
                if len(line) >= 4:
                    lines.append(tuple(line))
        return tuple(lines)

    def _coordinates(self, move):
        return move // STRIDE - 1, move % STRIDE - 1

    def _centre_distance(self, move):
        r, c = self._coordinates(move)
        return abs(r - SIZE // 2) + abs(c - SIZE // 2)

    def _check_time(self):
        if perf_counter() >= self._deadline:
            raise TimeoutError

import random
import sys
from pathlib import Path
from time import perf_counter

from smartplayerTeamName import SmartPlayerTeamName, CELLS, STRIDE, WALL

SIZE = 19
AXES = ((0, 1), (1, 0), (1, 1), (1, -1))


def grid():
    return [[-1] * SIZE for _ in range(SIZE)]


def has_five(board, color):
    for r in range(SIZE):
        for c in range(SIZE):
            if board[r][c] != color:
                continue
            for dr, dc in AXES:
                pr, pc = r - dr, c - dc
                if 0 <= pr < SIZE and 0 <= pc < SIZE and board[pr][pc] == color:
                    continue
                n, rr, cc = 0, r, c
                while 0 <= rr < SIZE and 0 <= cc < SIZE and board[rr][cc] == color:
                    n += 1
                    rr, cc = rr + dr, cc + dc
                if n == 5:
                    return True
    return False


def oracle(board, move, color):
    after = [row[:] for row in board]
    r, c = move
    after[r][c] = color
    removed = set()
    for dr, dc in AXES:
        for sign in (-1, 1):
            rr, cc = r + sign * 3 * dr, c + sign * 3 * dc
            if not (0 <= rr < SIZE and 0 <= cc < SIZE):
                continue
            a = (r + sign * dr, c + sign * dc)
            b = (r + sign * 2 * dr, c + sign * 2 * dc)
            if after[a[0]][a[1]] == after[b[0]][b[1]] == 1 - color and after[rr][cc] == color:
                removed.update((a, b))
    for rr, cc in removed:
        after[rr][cc] = -1
    mine, theirs = has_five(after, color), has_five(after, 1 - color)
    result = 0 if mine and theirs else 1 if mine else -1 if theirs else None
    return after, removed, result


def set_position(agent, board):
    agent._grid = [WALL] * (STRIDE * STRIDE)
    for r in range(SIZE):
        for c in range(SIZE):
            agent._grid[(r + 1) * STRIDE + c + 1] = board[r][c]
    agent._deadline = float("inf")


def check_simulation(agent, board, move, color):
    set_position(agent, board)
    original = agent._grid[:]
    p = (move[0] + 1) * STRIDE + move[1] + 1
    taken = agent._make_move(p, color)
    expected, removed, result = oracle(board, move, color)
    actual = [[agent._grid[(r + 1) * STRIDE + c + 1] for c in range(SIZE)] for r in range(SIZE)]
    assert actual == expected, (move, color, "capture mismatch")
    assert {agent._coordinates(q) for q in taken} == removed
    assert agent._outcome(p, color, taken) == result, (move, color, "outcome mismatch")
    agent._undo_move(p, color, taken)
    assert agent._grid == original, "undo mismatch"

def raw_wins(agent, color):
    result = set()
    for move in CELLS:
        if agent._grid[move] != -1:
            continue
        agent._grid[move] = color
        if agent._exact_five_at(move, color):
            result.add(move)
        agent._grid[move] = -1
    return result


def main():
    rng = random.Random(20261010)
    positions = 0
    for _ in range(100):
        board = grid()
        for index in rng.sample(range(361), rng.randrange(20, 230)):
            board[index // 19][index % 19] = rng.randrange(2)
        for color in (0, 1):
            agent = prepare(board, color)
            assert set(agent._potential_wins(color)) == raw_wins(agent, color)
            positions += 1

    # At the frontier, a side that can win immediately receives a win value.
    for color in (0, 1):
        board = grid()
        for c in range(5, 9):
            board[9][c] = color
        agent = prepare(board, color)
        assert agent._search(color, 0, -2 * MATE, 2 * MATE, 1) == MATE - 1

    # An open four has no defense in this isolated position.
    board = grid()
    for c in range(5, 9):
        board[9][c] = 0
    agent = prepare(board, 1)
    assert agent._search(1, 0, -2 * MATE, 2 * MATE, 1) == -MATE + 2

    # A capture prevents the immediate loss; later attacks may still win.
    board[8][6], board[10][6] = 0, 1
    agent = prepare(board, 1)
    defensive = {agent._coordinates(p) for p in agent._defending_moves(1)}
    assert (7, 6) in defensive
    assert agent._quiescence(1, -2 * MATE, 2 * MATE, 1, 1) > -MATE + 1000
    assert agent.last_extension > 0

    # A single block creates two new opposing winning moves by multi-capture.
    # Other capturing defenses prevent the old and newly created threats.
    board = grid()
    for c in range(2, 6):
        board[10][c] = 1
    board[10][1] = 0
    for c in range(2, 8):
        board[11][c] = 1
    board[12][6] = board[12][8] = 1
    board[13][6] = board[13][9] = board[12][5] = 0

    assert (9, 2) in safe and (9, 5) in safe
    ranked = agent._rank_moves(0)

    agent = prepare(board, 0)
    assert agent._quiescence(0, -2 * MATE, 2 * MATE, 1, QUIESCENCE_DEPTH) > MATE - 100

# AI usage: ChatGPT (Codex) wrote this local, cumulative-time match runner.
import argparse
import json
import random
import sys
from pathlib import Path
from time import perf_counter

sys.path.insert(0, str(Path(__file__).parent / "omok-agent"))
from smartplayerTeamName import SmartPlayerTeamName


def play(opponent_class, agent_color, seed, board_class):
    random.seed(seed)
    board = board_class()
    players = [None, None]
    players[agent_color] = SmartPlayerTeamName(agent_color)
    players[1 - agent_color] = opponent_class(1 - agent_color)
    remaining = [60.0, 60.0]
    turns = [0, 0]
    current = 0
    for ply in range(1200):
        before = board.GetBoard()
        snapshot = [row[:] for row in before]
        started = perf_counter()
        try:
            move = players[current].take_turn(before, remaining[current] * 1000)
        except Exception as exc:
            return {"result": "loss" if current == agent_color else "win", "reason": f"player {current} exception: {type(exc).__name__}: {exc}", "plies": ply}
        remaining[current] -= perf_counter() - started
        turns[current] += 1
        if remaining[current] < 0 or before != snapshot:
            return {"result": "loss" if current == agent_color else "win", "reason": "timeout or input mutation", "plies": ply}
        if not (isinstance(move, tuple) and len(move) == 2 and all(type(x) is int for x in move) and board.ValidPosition(*move)):
            return {"result": "loss" if current == agent_color else "win", "reason": "illegal move", "plies": ply}
        board.PlaceStone(*move, current)
        board.TakeStone(*move, current)
        wins = [board.CheckOmok(color) for color in (0, 1)]
        result = "draw" if all(wins) else "win" if wins[agent_color] else "loss" if wins[1 - agent_color] else "draw" if board.FullBoard() else None
        if result:
            return {"result": result, "reason": "board result", "plies": ply + 1, "agent_seconds": round(60 - remaining[agent_color], 4), "opponent_seconds": round(60 - remaining[1 - agent_color], 4), "agent_turns": turns[agent_color]}
        current = 1 - current
    return {"result": "unfinished", "reason": "testing ply cap (not a game draw)", "plies": 1200}


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("existing_folder", help="Folder containing the original board.py, player.py, randomplayer.py, simpleplayer.py")
    parser.add_argument("--games", type=int, default=8)
    parser.add_argument("--opponent", choices=("random", "simple", "both"), default="both")
    parser.add_argument("--output")
    args = parser.parse_args()
    sys.path.insert(0, str(Path(args.existing_folder).resolve()))
    from board import Board
    from randomplayer import RandomPlayer
    from simpleplayer import SimplePlayer
    results = []
    opponents = [("RandomPlayer", RandomPlayer), ("SimplePlayer", SimplePlayer)]
    if args.opponent != "both":
        opponents = [opponents[0 if args.opponent == "random" else 1]]
    for name, opponent in opponents:
        for game in range(args.games):
            color = game % 2
            result = play(opponent, color, 2026100600 + game, Board)
            result.update(opponent=name, agent_color="black" if color == 0 else "white", seed=2026100600 + game)
            results.append(result)
            print(json.dumps(result), flush=True)
    payload = {"limit_seconds_per_player_per_game": 60, "games": results}
    if args.output:
        Path(args.output).write_text(json.dumps(payload, indent=2), encoding="utf-8")


if __name__ == "__main__":
    main()

    parser = argparse.ArgumentParser()

    board.PrintBoard()
    while True:
        supplied = board.GetBoard()
        snapshot = [row[:] for row in supplied]
        started = perf_counter()
        r, c = players[current].take_turn(supplied, -1 if args.minutes == -1 else remaining[current] * 1000)
        elapsed = perf_counter() - started
        if args.minutes != -1:
            remaining[current] -= elapsed
            if remaining[current] < 0:
                print("Time Out!", "White Wins" if current == 0 else "Black Wins")

        board.PlaceStone(r, c, current)
        board.TakeStone(r, c, current)
        board.PrintBoard()
        wins = [board.CheckOmok(color) for color in (0, 1)]
        if all(wins) or board.FullBoard() and not any(wins):
            print("Draw")
            return
        if any(wins):
            print("Black Wins" if wins[0] else "White Wins")
            return
        current = 1 - current

3. 이번 변경
일반 탐색의 최대 목표 깊이: 4수 -> 6수.
일반 후보 폭: 첫 수 14개 -> 20개, 이후 10개 -> 12개.
즉시 승리를 만드는 공격 후보는 일반 탐색의 개수 제한에 더해 포함합니다.
상대의 다음 수 승리를 막을 수 있는 방어 후보는 개수 제한 없이 검토합니다.
상대의 포획 공격을 해제하는 방어와, 내 포획으로 새로 생긴 상대 위협도 계산합니다.
탐색 끝에서도 즉시 승리, 상대의 즉시 승리, 가능한 방어를 확인합니다.

전술 연장의 공격 후보는 점수 순 최대 8개이며, 필수 방어 후보는 모두 포함합니다.
첫 탐색 단계가 시간 내 끝나지 않아도 이미 계산한 후보 결과를 활용합니다.
첫 단계에서 전술 연장으로 승리 값이 확인되면 해당 수를 반환할 수 있습니다.

이번 버전은 앞서 분석한 네 기준 중 **탐색 폭·깊이**를 강화합니다. 후보 생성 범위, 평가 점수, 시간 배분 공식과 게임 규칙은 유지합니다.

## 전과 후

| 부분 | 초안 | 강화 버전 |
|---|---|---|
| 일반 탐색의 최대 목표 깊이 | 4수 | 6수 |
| 첫 수의 일반 후보 수 | 14개 | 20개 + 중요한 공격 후보 |
| 이후의 일반 후보 수 | 10개 | 12개 + 중요한 공격 후보 |
| 필수 방어 | 기존 위협을 막는 후보 정렬 후 개수 제한 | 새 위협까지 확인하고 가능한 방어 모두 포함 |
| 깊이 끝에서의 처리 | 판 점수 계산 | 즉시 승리·방어 확인 뒤 전술 연장 |
| 전술 연장 | 없음 | 공격/방어 연속 최대 6수 추가 |

수는 한 번의 착수를 뜻합니다. 일반 탐색 3수는 내 착수, 상대 착수, 내 착수입니다. 최대 깊이는 시도하는 상한이며 매 차례 도달하는 깊이가 아닙니다.

## 왜 연장하는가

앞을 세 수 읽었더라도 그 세 번째 수가 강한 공격을 시작하는 수라면, 그 직후 판의 모양만 보고 판단하기 어렵습니다. 상대가 어떻게 막고, 내가 이어서 어떻게 공격할 수 있는지 확인할 필요가 있습니다.

새 `_quiescence()`는 깊이 끝에서 다음 순서로 판단합니다.

1. 현재 차례의 플레이어에게 즉시 승리 수가 있으면 승리 값을 반환합니다.
2. 상대에게 즉시 승리 수가 있으면, 착수와 포획을 적용한 뒤 그 승리를 막는 수를 모두 찾습니다.
3. 막을 수 있는 수가 없으면 다음 차례 패배로 평가합니다.
4. 방어가 가능하면 그 응수와 후속 상황을 더 읽습니다.
5. 즉시 위협이 없는 판에서는 판 점수를 계산하고, 다음 차례 승리를 위협하는 공격 후보를 추가로 읽습니다.
6. 전술 연장의 상한이나 시간 예산에 도달하면 계산을 끝냅니다.

이것을 퀴에슨스 탐색이라고 부릅니다. 중요한 연속 공격/방어에서 평가를 잠시 미루고 더 읽는 방법입니다. 여기서는 과제의 포획과 정확히 다섯 개 규칙에 맞춰 구현했습니다.

## 포획 때문에 새 위협이 생기는 방어

현재 상대의 승리 지점을 막았어도, 동시에 돌을 잡아 상대의 긴 줄을 네 돌로 줄이면 다른 승리 지점이 생길 수 있습니다. 새 버전은 가상 방어 후의 판에서 상대 승리 수를 다시 찾습니다. 기존 위협의 좌표가 막혔다는 것만으로 안전하다고 판단하지 않습니다.

`test_search.py`에는 이 상황을 넣었습니다. 겉보기에는 막는 수인 `(10, 6)`이 다중 포획으로 다른 위협을 만들어 제외되고, 포획으로 전체 위협을 없애는 `(9, 2)`, `(9, 5)`가 남는지 검사합니다. 좌표는 모두 0부터 셉니다.

## 탐색 폭 확대

일반 후보 수를 늘렸고, 다음 수에 승리할 가능성을 만드는 공격은 그 제한 바깥에서도 추가합니다. 상대의 즉시 승리를 막는 방어는 모두 검토합니다.

전술 연장의 공격은 점수 순 최대 8개입니다. 연장을 모든 후보에 적용하면 짧은 시간에 지나치게 많은 계산을 하게 되기 때문입니다. 필수 방어에서는 이 8개 제한을 적용하지 않습니다.

## 시간이 끝났을 때

완료한 일반 탐색 단계가 있으면 그 결과를 사용합니다. 첫 단계가 아직 끝나지 않았더라도 응수 계산을 마친 후보가 있으면 그중 가장 좋은 결과를 활용합니다. 가상 착수는 `finally`에서 되돌리므로 계산이 중단돼도 입력 판과 개인 복사본이 오염되지 않습니다.

`last_depth`는 일반 탐색의 완료/선택 단계, `last_qnodes`는 전술 연장에서 방문한 지점 수, `last_extension`은 이번 착수에서 시도한 최대 연장 길이를 볼 때 사용할 수 있습니다. 시도한 최대 길이가 모든 후보에 적용되었다는 뜻은 아닙니다.

## 변경 효과를 해석하는 방법

전술 회귀 검사는 새 기능이 의도대로 작동하는지 확인합니다. 봇 대결은 그 기능이 전체 경기에서 얼마나 도움 되는지 확인합니다. 두 결과를 함께 봐야 합니다.

시간 제한이 있는 탐색이므로 컴퓨터 속도에 따라 완료 단계와 선택이 바뀔 수 있습니다. 이번 성적은 현재 실행 환경의 개발 결과이며 TA 상대 성적이나 우승을 보장하지 않습니다. 다음 개선에서는 실제 패배 기록을 계속 모으고 선택적 탐색의 누락과 계산 비용을 확인할 수 있습니다.
