from time import perf_counter
from collections import OrderedDict

from playerJangJinha import PlayerJangJinha

SIZE = 19
EMPTY = -1
WALL = 2
STRIDE = SIZE + 2
AXES = (1, STRIDE, STRIDE + 1, STRIDE - 1)
DIRECTIONS = AXES + tuple(-d for d in AXES)
CELLS = tuple((r + 1) * STRIDE + c + 1 for r in range(SIZE) for c in range(SIZE))
MATE = 10_000_000
WINDOW_WEIGHTS = (0, 2, 30, 450, 8_000, 0)
MAX_DEPTH = 6
QUIESCENCE_DEPTH = 6
ROOT_WIDTH = 20
INNER_WIDTH = 12
FORCING_WIDTH = 8
POSITION_CACHE_LIMIT = 20_000
LINE_CACHE_LIMIT = 40_000


class SmartPlayerJangJinha(PlayerJangJinha):
    def __init__(self, color):
        super().__init__(color)
        if color not in (0, 1):
            raise ValueError("color must be 0 (black) or 1 (white)")
        self._lines = self._build_lines()
        neighbors = [[] for _ in range(STRIDE * STRIDE)]
        for p in CELLS:
            r, c = self._coordinates(p)
            neighbors[p] = tuple((rr + 1) * STRIDE + cc + 1
                                 for rr in range(max(0, r - 2), min(SIZE, r + 3))
                                 for cc in range(max(0, c - 2), min(SIZE, c + 3)))
        self._neighbors = tuple(neighbors)
        self._indexed_grid = None
        self._line_cache = OrderedDict()
        self.last_depth = 0
        self.last_nodes = 0
        self.last_qnodes = 0
        self.last_extension = 0
        self._tactical_cache = {}
        self._defence_cache = {}
        self._forcing_cache = {}
        self.last_tt_hits = 0
        self.last_rank_hits = 0
        self.last_move_score_hits = 0
        self.last_winning_hits = 0
        self.last_line_hits = 0
        self.last_make_calls = 0

    def take_turn(self, board, time):
        started = perf_counter()
        self._grid = [WALL] * (STRIDE * STRIDE)
        for r in range(SIZE):
            for c in range(SIZE):
                self._grid[(r + 1) * STRIDE + c + 1] = int(board[r][c])
        empty = [p for p in CELLS if self._grid[p] == EMPTY]
        if not empty:
            raise ValueError("No legal move: the board is full.")
        if len(empty) == SIZE * SIZE:
            return SIZE // 2, SIZE // 2

        candidates = self._candidates()
        best = min(candidates, key=self._centre_distance)
        remaining = float(time) / 1000 if time != -1 else None
        if remaining is not None and remaining <= 0.015:
            return self._coordinates(best)
        budget = 1.0 if remaining is None else min(1.0, remaining / 30, max(0, remaining - 0.02))
        self._deadline = started + budget
        self.last_depth = 0
        self.last_nodes = 0
        self.last_qnodes = 0
        self.last_extension = 0
        self.last_tt_hits = 0
        self.last_rank_hits = 0
        self.last_move_score_hits = 0
        self.last_winning_hits = 0
        self.last_line_hits = 0
        self.last_make_calls = 0

        try:
            ranked = self._rank_moves(self.color)
            if ranked:
                best = ranked[0][1]
            if ranked and ranked[0][0] >= MATE:
                return self._coordinates(best)
            for depth in range(1, MAX_DEPTH + 1):
                selected = self._select_moves(ranked, self.color, ROOT_WIDTH)
                ordered = sorted(selected, key=lambda item: item[1] != best)
                depth_best, depth_score = best, -2 * MATE
                alpha = -2 * MATE
                for _, move in ordered:
                    self._check_time()
                    captured = self._make_move(move, self.color)
                    try:
                        outcome = self._outcome(move, self.color, captured)
                        if outcome is not None:
                            score = outcome * MATE
                        else:
                            score = -self._search(1 - self.color, depth - 1, -2 * MATE, -alpha, 1)
                    finally:
                        self._undo_move(move, self.color, captured)
                    if score > depth_score:
                        depth_best, depth_score = move, score
                        if depth == 1:
                            best = depth_best
                    alpha = max(alpha, score)
                    if depth == 1 and score > MATE - 1000:
                        self.last_depth = 1
                        return self._coordinates(move)
                best = depth_best
                self.last_depth = depth
        except TimeoutError:
            pass
        return self._coordinates(best)

    def _search(self, color, depth, alpha, beta, ply):
        self._check_time()
        self._ensure_position()
        self.last_nodes += 1
        key = (self._position_key(color), depth, ply)
        cached = self._search_cache.get(key)
        if cached is not None:
            value, flag = cached
            if flag == 0 or flag == 1 and value >= beta or flag == -1 and value <= alpha:
                self.last_tt_hits += 1
                return value
        value = self._search_uncached(color, depth, alpha, beta, ply)
        self._remember(self._search_cache, key, (value, self._bound(value, alpha, beta)))
        return value

    def _search_uncached(self, color, depth, alpha, beta, ply):
        if self._empty_count == 0:
            return 0
        if depth == 0:
            return self._quiescence(color, alpha, beta, ply, QUIESCENCE_DEPTH)
        wins, draws, threats = self._tactical_state(color)
        if wins:
            return MATE - ply
        if threats and not self._defending_moves(color):
            return -MATE + ply + 1
        moves = self._rank_moves(color)
        if moves[0][0] >= MATE:
            return MATE - ply
        best = -2 * MATE
        for _, move in self._select_moves(moves, color, INNER_WIDTH):
            self._check_time()
            captured = self._make_move(move, color)
            try:
                outcome = self._outcome(move, color, captured)
                if outcome is not None:
                    score = outcome * (MATE - ply)
                else:
                    score = -self._search(1 - color, depth - 1, -beta, -alpha, ply + 1)
            finally:
                self._undo_move(move, color, captured)
            best = max(best, score)
            alpha = max(alpha, score)
            if alpha >= beta:
                break
        return best

    def _rank_moves(self, color):
        self._check_time()
        key = self._position_key(color)
        cached = self._rank_cache.get(key)
        if cached is not None:
            self.last_rank_hits += 1
            ranked, forcing = cached
            self._remember(self._forcing_cache, key, forcing)
            return list(ranked)
        wins, draws, threats = self._tactical_state(color)
        if wins:
            ranked = [(MATE, move) for move in wins]
            self._remember(self._rank_cache, key, (tuple(ranked), ()))
            return ranked
        candidates = self._candidates()
        if threats:
            candidates = self._defending_moves(color) or candidates
        ranked = self._score_moves(candidates, color)
        self._remember(self._rank_cache, key, (tuple(ranked), self._forcing_cache[key]))
        return ranked

    def _score_moves(self, candidates, color):
        opponent = 1 - color
        result, forcing = [], []
        key = self._position_key(color)
        scored = self._move_score_cache.get(key)
        if scored is None:
            scored = {}
            self._remember(self._move_score_cache, key, scored)
        for move in candidates:
            self._check_time()
            cached = scored.get(move)
            if cached is not None:
                self.last_move_score_hits += 1
                score, is_forcing = cached
                result.append((score, move))
                if is_forcing:
                    forcing.append(move)
                continue
            is_forcing = False
            captured = self._make_move(move, color)
            try:
                outcome = self._outcome(move, color, captured)
                local = self._local_score(move, color)
                attack = local + 600 * len(captured)
                attack -= 900 * self._vulnerable_pairs(move, color)
                if outcome == 0:
                    is_forcing = True
                elif outcome is None and (local >= 8_000 or captured):
                    if self._winning_options(color)[0]:
                        is_forcing = True
            finally:
                self._undo_move(move, color, captured)
            captured = self._make_move(move, opponent)
            try:
                defence = self._local_score(move, opponent) + 600 * len(captured)
            finally:
                self._undo_move(move, opponent, captured)
            score = attack + 0.95 * defence - self._centre_distance(move) * 0.1
            if outcome == -1:
                score = -MATE
            elif outcome == 0:
                score = -50
            elif outcome == 1:
                score = MATE
            scored[move] = score, is_forcing
            if is_forcing:
                forcing.append(move)
            result.append((score, move))
        self._remember(self._forcing_cache, key, tuple(forcing))
        return sorted(result, reverse=True)

    def _select_moves(self, ranked, color, width):
        if self._tactical_state(color)[2]:
            return ranked
        forcing = set(self._forcing_cache.get(self._position_key(color), ()))
        selected = list(ranked[:width])
        selected.extend(item for item in ranked[width:] if item[1] in forcing)
        return selected

    def _position_key(self, color):
        self._ensure_position()
        if self._board_key is None:
            self._board_key = bytes(self._state)
        return color, self._board_key

    def _potential_wins(self, color):
        key = self._position_key(color)
        cached = self._potential_cache.get(key)
        if cached is not None:
            return cached
        grid, result = self._grid, set()
        for move in self._stones[color]:
            if grid[move] != color:
                continue
            for axis in AXES:
                if grid[move - axis] == color:
                    continue
                length, end = 0, move
                while grid[end] == color:
                    length += 1
                    end += axis
                if length == 4:
                    before = move - axis
                    if grid[before] == EMPTY and grid[before - axis] != color:
                        result.add(before)
                    if grid[end] == EMPTY and grid[end + axis] != color:
                        result.add(end)
                elif length < 4 and grid[end] == EMPTY:
                    tail, count = end + axis, 0
                    while grid[tail] == color:
                        count += 1
                        tail += axis
                    if length + count == 4:
                        result.add(end)
        result = tuple(sorted(result))
        self._remember(self._potential_cache, key, result)
        return result

    def _winning_options(self, color):
        self._check_time()
        key = self._position_key(color)
        cached = self._winning_cache.get(key)
        if cached is not None:
            self.last_winning_hits += 1
            return cached
        wins, draws = [], []
        for move in self._potential_wins(color):
            self._check_time()
            captured = self._make_move(move, color)
            try:
                outcome = self._outcome(move, color, captured)
                if outcome == 1:
                    wins.append(move)
                elif outcome == 0:
                    draws.append(move)
            finally:
                self._undo_move(move, color, captured)
        result = tuple(wins), tuple(draws)
        self._remember(self._winning_cache, key, result)
        return result

    def _tactical_state(self, color):
        self._check_time()
        key = self._position_key(color)
        if key not in self._tactical_cache:
            wins, draws = self._winning_options(color)
            threats = self._winning_options(1 - color)[0]
            self._remember(self._tactical_cache, key, (wins, draws, threats))
        return self._tactical_cache[key]

    def _defending_moves(self, color):
        key = self._position_key(color)
        if key in self._defence_cache:
            return self._defence_cache[key]
        safe = []
        for move in self._candidates():
            self._check_time()
            captured = self._make_move(move, color)
            try:
                outcome = self._outcome(move, color, captured)
                if outcome in (0, 1):
                    safe.append(move)
                elif outcome is None and not self._winning_options(1 - color)[0]:
                    safe.append(move)
            finally:
                self._undo_move(move, color, captured)
        self._remember(self._defence_cache, key, tuple(safe))
        return self._defence_cache[key]

    def _forcing_candidates(self, color):
        grid, near = self._grid, set()
        for line in self._lines:
            self._check_time()
            for start in range(len(line) - 4):
                cells = line[start:start + 5]
                values = [grid[p] for p in cells]
                if values.count(color) == 3 and values.count(EMPTY) == 2:
                    near.update(p for p in cells if grid[p] == EMPTY)
        for move in CELLS:
            if grid[move] != color:
                continue
            for direction in DIRECTIONS:
                end = move + 3 * direction
                if 0 <= end < len(grid) and grid[end] == EMPTY:
                    if grid[move + direction] == grid[move + 2 * direction] == 1 - color:
                        near.add(end)
        for move in self._potential_wins(color):
            for direction in DIRECTIONS:
                end = move + 3 * direction
                if 0 <= end < len(grid) and grid[end] == EMPTY:
                    if grid[move + direction] == grid[move + 2 * direction] == 1 - color:
                        near.add(end)
        return tuple(sorted(near))

    def _forcing_moves(self, color):
        key = self._position_key(color)
        if key in self._forcing_cache:
            return self._score_moves(self._forcing_cache[key], color)
        forcing = []
        for move in self._forcing_candidates(color):
            self._check_time()
            captured = self._make_move(move, color)
            try:
                outcome = self._outcome(move, color, captured)
                if outcome == 0 or outcome == 1:
                    forcing.append(move)
                elif outcome is None and self._winning_options(color)[0]:
                    forcing.append(move)
            finally:
                self._undo_move(move, color, captured)
        return self._score_moves(forcing, color)

    def _quiescence(self, color, alpha, beta, ply, remaining):
        self._check_time()
        self._ensure_position()
        self.last_qnodes += 1
        self.last_extension = max(self.last_extension, QUIESCENCE_DEPTH - remaining)
        key = (self._position_key(color), remaining, ply)
        cached = self._q_cache.get(key)
        if cached is not None:
            value, flag = cached
            if flag == 0 or flag == 1 and value >= beta or flag == -1 and value <= alpha:
                self.last_tt_hits += 1
                return value
        value = self._quiescence_uncached(color, alpha, beta, ply, remaining)
        self._remember(self._q_cache, key, (value, self._bound(value, alpha, beta)))
        return value

    def _quiescence_uncached(self, color, alpha, beta, ply, remaining):
        if self._empty_count == 0:
            return 0
        wins, draws, threats = self._tactical_state(color)
        if wins:
            return MATE - ply
        if threats:
            safe = self._defending_moves(color)
            if not safe:
                return -MATE + ply + 1
            if remaining == 0:
                score = self._evaluate(color)
                return max(score, 0) if draws else score
            moves = self._score_moves(safe, color)
            best = -2 * MATE
        else:
            best = self._evaluate(color)
            if draws:
                best = max(best, 0)
            if remaining == 0 or best >= beta:
                return best
            alpha = max(alpha, best)
            moves = self._forcing_moves(color)[:FORCING_WIDTH]
        for _, move in moves:
            self._check_time()
            captured = self._make_move(move, color)
            try:
                outcome = self._outcome(move, color, captured)
                if outcome is not None:
                    score = outcome * (MATE - ply)
                else:
                    score = -self._quiescence(1 - color, -beta, -alpha, ply + 1, remaining - 1)
            finally:
                self._undo_move(move, color, captured)
            best = max(best, score)
            alpha = max(alpha, score)
            if alpha >= beta:
                break
        return best

    def _candidates(self):
        key = self._position_key(0)[1]
        cached = self._candidate_cache.get(key)
        if cached is None:
            near = {q for side in self._stones for p in side for q in self._neighbors[p]
                    if self._grid[q] == EMPTY}
            cached = tuple(sorted(near)) or tuple(p for p in CELLS if self._grid[p] == EMPTY)
            self._remember(self._candidate_cache, key, cached)
        return list(cached)

    def _make_move(self, move, color):
        self._ensure_position()
        previous_key = self._board_key
        self.last_make_calls += 1
        grid = self._grid
        grid[move] = color
        opponent = 1 - color
        captured = []
        for direction in DIRECTIONS:
            a, b, end = move + direction, move + 2 * direction, move + 3 * direction
            if 0 <= end < len(grid) and grid[a] == opponent and grid[b] == opponent and grid[end] == color:
                captured.extend((a, b))
        for p in captured:
            grid[p] = EMPTY
        self._stones[color].add(move)
        self._state[move] = color + 1
        for p in captured:
            self._stones[1 - color].remove(p)
            self._state[p] = 0
        self._empty_count += len(captured) - 1
        self._board_key = None
        self._move_stack.append(previous_key)
        return captured

    def _undo_move(self, move, color, captured):
        self._board_key = self._move_stack.pop()
        self._grid[move] = EMPTY
        self._state[move] = 0
        self._stones[color].remove(move)
        self._empty_count += 1 - len(captured)
        for p in captured:
            self._grid[p] = 1 - color
            self._state[p] = 2 - color
            self._stones[1 - color].add(p)

    def _exact_five_at(self, move, color):
        if self._grid[move] != color:
            return False
        for axis in AXES:
            length = 1
            for direction in (axis, -axis):
                p = move + direction
                while 0 <= p < len(self._grid) and self._grid[p] == color:
                    length += 1
                    p += direction
            if length == 5:
                return True
        return False

    def _outcome(self, move, color, captured):
        mine = self._exact_five_at(move, color)
        opponent = 1 - color
        theirs = False
        for removed in captured:
            for direction in DIRECTIONS:
                p = removed + direction
                if 0 <= p < len(self._grid) and self._grid[p] == opponent:
                    if self._exact_five_at(p, opponent):
                        theirs = True
                        break
            if theirs:
                break
        if mine and theirs:
            return 0
        if mine:
            return 1
        if theirs:
            return -1
        return None

    def _local_score(self, move, color):
        grid = self._grid
        total = 0
        for axis in AXES:
            length, open_ends = 1, 0
            for direction in (axis, -axis):
                p = move + direction
                while 0 <= p < len(grid) and grid[p] == color:
                    length += 1
                    p += direction
                if 0 <= p < len(grid) and grid[p] == EMPTY:
                    open_ends += 1
            if length == 4:
                total += 45_000 if open_ends == 2 else 6_000 if open_ends else 0
            elif length == 3:
                total += 3_000 if open_ends == 2 else 250 if open_ends else 0
            elif length == 2:
                total += 120 if open_ends == 2 else 20 if open_ends else 0
            for offset in range(5):
                start = move - offset * axis
                end = start + 4 * axis
                if not (0 <= start < len(grid) and 0 <= end < len(grid)):
                    continue
                cells = [grid[start + i * axis] for i in range(5)]
                if any(value not in (EMPTY, color) for value in cells):
                    continue
                before, after = start - axis, end + axis
                if 0 <= before < len(grid) and grid[before] == color:
                    continue
                if 0 <= after < len(grid) and grid[after] == color:
                    continue
                total += WINDOW_WEIGHTS[cells.count(color)]
        return total

    def _vulnerable_pairs(self, move, color):
        count, grid = 0, self._grid
        for direction in DIRECTIONS:
            a, before, after = move + direction, move - direction, move + 2 * direction
            if not (0 <= before < len(grid) and 0 <= after < len(grid)):
                continue
            if grid[a] == color and {grid[before], grid[after]} == {1 - color, EMPTY}:
                count += 1
        return count

    def _evaluate(self, color):
        key = self._position_key(0)[1]
        total = self._eval_cache.get(key)
        if total is None:
            total = 0
            for line in self._lines:
                values = tuple(self._grid[p] for p in line)
                score = self._line_cache.get(values)
                if score is None:
                    score = self._score_line(values, 0) - self._score_line(values, 1)
                    self._remember(self._line_cache, values, score, LINE_CACHE_LIMIT)
                else:
                    self.last_line_hits += 1
                total += score
            self._remember(self._eval_cache, key, total)
        return total if color == 0 else -total

    @staticmethod
    def _bound(value, alpha, beta):
        return -1 if value <= alpha else 1 if value >= beta else 0

    @staticmethod
    def _remember(cache, key, value, limit=POSITION_CACHE_LIMIT):
        if key not in cache and len(cache) >= limit:
            if isinstance(cache, OrderedDict):
                cache.popitem(last=False)
            else:
                cache.pop(next(iter(cache)))
        cache[key] = value

    def _ensure_position(self):
        if self._indexed_grid is self._grid:
            return
        self._indexed_grid = self._grid
        self._state = bytearray(value + 1 for value in self._grid)
        self._board_key = None
        self._stones = [{p for p in CELLS if self._grid[p] == color} for color in (0, 1)]
        self._empty_count = SIZE * SIZE - sum(map(len, self._stones))
        self._move_stack = []
        for name in ('_tactical_cache', '_defence_cache', '_forcing_cache',
                     '_winning_cache', '_potential_cache', '_rank_cache', '_move_score_cache',
                     '_candidate_cache', '_search_cache', '_q_cache', '_eval_cache'):
            setattr(self, name, OrderedDict())




    def _score_line(self, values, color):
        total = 0
        for start in range(len(values) - 4):
            window = values[start:start + 5]
            if any(value not in (EMPTY, color) for value in window):
                continue
            if start and values[start - 1] == color:
                continue
            if start + 5 < len(values) and values[start + 5] == color:
                continue
            count = window.count(color)
            total += WINDOW_WEIGHTS[count]
        i = 0
        while i < len(values):
            if values[i] != color:
                i += 1
                continue
            start = i
            while i < len(values) and values[i] == color:
                i += 1
            length = i - start
            ends = int(start > 0 and values[start - 1] == EMPTY) + int(i < len(values) and values[i] == EMPTY)
            if length == 4 and ends == 2:
                total += 30_000
            elif length == 3 and ends == 2:
                total += 2_000
            elif length == 2 and ends == 2:
                total += 80
        for i in range(len(values) - 3):
            pair = values[i:i + 4]
            if pair in ((1 - color, color, color, EMPTY), (EMPTY, color, color, 1 - color)):
                total -= 700
        return total

    def _build_lines(self):
        lines = []
        for axis in AXES:
            for p in CELLS:
                r, c = self._coordinates(p - axis)
                if 0 <= r < SIZE and 0 <= c < SIZE:
                    continue
                line, q = [], p
                while True:
                    r, c = self._coordinates(q)
                    if not (0 <= r < SIZE and 0 <= c < SIZE):
                        break
                    line.append(q)
                    q += axis
                if len(line) >= 4:
                    lines.append(tuple(line))
        return tuple(lines)

    def _coordinates(self, move):
        return move // STRIDE - 1, move % STRIDE - 1

    def _centre_distance(self, move):
        r, c = self._coordinates(move)
        return abs(r - SIZE // 2) + abs(c - SIZE // 2)

    def _check_time(self):
        if perf_counter() >= self._deadline:
            raise TimeoutError

    classes = {'v2': load('comparison_v2', ROOT / 'verification/reference_v2/smartplayerJangJinha.py'),

                              'position_count': len(positions), 'python': sys.version,
                              'platform': platform.platform(),
                              'notes': 'Positions from the saved 2026-10-10 tournament. Measures search performance, not new match wins. Alternate v2/v3 execution order; tactical extensions are separate from normal depth.'},
               'status': 'running', 'records': records}
    for repeat in range(2):

                started = perf_counter()

                assert isinstance(move, tuple) and all(type(x) is int for x in move)

                result = {'repeat': repeat + 1, 'game_id': position['game_id'],
                          'before_ply': position['before_ply'], 'version': version,

                          'normal_depth': agent.last_depth, 'normal_nodes': agent.last_nodes,
                          'quiescence_nodes': agent.last_qnodes, 'max_tactical_extension': agent.last_extension}
                if version == 'v3':

# JangJinha v3: 계산 재사용

v3는 v2의 평가 기준·후보 범위·탐색 폭과 최대 깊이·시간 배분을 유지하며 반복 계산을 줄입니다.

## 재사용하는 정보

- 같은 판의 후보 좌표, 즉시 승리/무승부 후보, 안전한 방어 후보.
- 같은 판과 색상의 착수별 점수와 전술 공격 분류, 정렬된 후보 목록.
- 같은 판의 평가 점수와 일반/전술 탐색 결과.
- 줄의 돌 배열에 대한 평가 점수. 이 결과는 차례가 바뀌어도 재사용합니다.

보드 식별에는 실제 441칸 센티널 배열을 그대로 인코딩한 bytes를 사용합니다.
확률적인 해시 일치만으로 다른 판을 같은 판으로 판단하지 않습니다.
돌 위치의 두 집합과 bytearray를 착수/포획/되돌리기 때 갱신하여
후보와 승리 좌표를 구할 때 빈칸까지 전부 확인하는 반복 작업을 줄입니다.
주변 2칸 좌표는 생성자에서 한 번 만들어 재사용합니다.

## 탐색 결과를 재사용하는 조건

색상, 동일한 판, 정확히 같은 일반 탐색 깊이/전술 연장 길이,
현재 착수 거리가 모두 같은 경우만 재사용합니다.
착수 거리를 구분하는 이유는 승패 점수에 승리까지의 수가 포함되기 때문입니다.

저장 값은 정확한 값, 하한, 상한을 구분합니다. 하한은 beta 이상일 때,
상한은 alpha 이하일 때만 검색 중단에 사용합니다.
더 깊은 검색 값을 얕은 검색의 정확한 값으로 대체하지 않습니다.
시간 초과 예외로 끝난 검색은 부모 위치의 완료 결과로 저장하지 않습니다.

일반 탐색과 전술 연장의 캐시를 분리합니다.
정렬 캐시를 다시 읽을 때 그때 계산한 전술 공격 목록도 함께 복구하여
후보 개수 제한 밖의 공격 수를 누락하지 않게 했습니다.

## 저장 범위

위치별 캐시는 새로운 take_turn 판을 준비할 때 초기화합니다.
판 평가에 쓰는 줄 배열 캐시는 계산 규칙이 같은 동안 다음 차례에도 유지합니다.
각 위치별 캐시는 최대 20,000개, 줄 점수 캐시는 최대 40,000개로 제한합니다.
오래된 항목을 제거하며, 제거는 정확도에 영향을 주지 않고 다시 계산하는 비용만 발생합니다.

## 검사

기존 규칙/탐색 검사에 추가하여 v2와 착수·포획·복구 상태를 800회 비교했습니다.
같은 위치·깊이의 전체 탐색 값, 창의 상한/하한, 색상과 전술 연장 길이,
승패 점수의 착수 거리, 판 변경과 취소 복구를 검사했습니다.
현재 코드의 판정·평가·공격 후보 선택 관련 9개 메서드와 기본 클래스,
시간 예산 및 탐색 설정이 v2와 같음을 비교했습니다.

동일 시간 비교는 같은 저장 위치 10개에 두 번씩 수행했습니다.
실행 순서는 v2/v3와 v3/v2를 교대했고, 매번 새 에이전트를 만들었습니다.
